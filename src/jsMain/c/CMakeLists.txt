cmake_minimum_required(VERSION 3.20)
project(trikeshed_uring_node LANGUAGES C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "The Node io_uring addon requires Linux")
endif()
find_path(NODE_INCLUDE_DIR node_api.h PATH_SUFFIXES node REQUIRED)
set(URING_INCLUDE_DIR "" CACHE STRING "Configured and source vendored liburing include directories")
set(URING_LIBRARY "" CACHE FILEPATH "Configured vendored liburing PIC static library")
find_path(URING_HEADER_DIR liburing.h PATHS ${URING_INCLUDE_DIR} NO_DEFAULT_PATH REQUIRED)
if(NOT EXISTS "${URING_LIBRARY}")
    message(FATAL_ERROR "URING_INCLUDE_DIR and URING_LIBRARY must identify the built vendored liburing")
endif()
add_library(trikeshed_uring MODULE uring_node.c)
target_include_directories(trikeshed_uring PRIVATE "${NODE_INCLUDE_DIR}" "${URING_INCLUDE_DIR}")
target_link_libraries(trikeshed_uring PRIVATE "${URING_LIBRARY}")
target_compile_features(trikeshed_uring PRIVATE c_std_11)
target_compile_definitions(trikeshed_uring PRIVATE NAPI_VERSION=8 _GNU_SOURCE)
target_compile_options(trikeshed_uring PRIVATE -Wall -Wextra -Werror)
set_target_properties(trikeshed_uring PROPERTIES PREFIX "" SUFFIX ".node")
