const {chromium}=require('playwright');const fs=require('fs'),path=require('path'),assert=require('assert/strict');
const out=process.argv[2],base=process.argv[3];let browser,context;const report={requests:[],errors:[]};
(async()=>{browser=await chromium.launch({executablePath:'/usr/bin/chromium',headless:true,args:['--no-sandbox','--disable-dev-shm-usage']});context=await browser.newContext({viewport:{width:1800,height:1200}});const page=await context.newPage();page.on('pageerror',e=>report.errors.push(String(e)));context.on('request',r=>{if(r.method()==='POST')report.requests.push({url:r.url(),body:r.postData()});});await page.goto(base+'/panels?load=airgap-document',{waitUntil:'domcontentloaded'});await page.waitForFunction(()=>document.querySelectorAll('.node').length===5&&document.body.innerText.includes('Loaded for review · choose Run or Live to execute'));await page.locator('#panelName').fill('airgap-coinbase-suski');const response=page.waitForResponse(r=>new URL(r.url()).pathname==='/api/lcnc/run'&&r.request().method()==='POST',{timeout:120000});await page.locator('#runBtn').click();report.receipt=await(await response).json();assert.equal(report.receipt.status,'completed');assert.equal(report.receipt.ok,true);assert.equal(report.errors.length,0);await context.request.post(base+'/__proof/finish');})().catch(e=>{report.error=String(e.stack||e);process.exitCode=1;}).finally(async()=>{if(context)await context.close();if(browser)await browser.close();fs.writeFileSync(path.join(out,'browser.json'),JSON.stringify(report,null,2));});
