def utf16(text,begin,end):return text.encode('utf-16-le')[begin*2:end*2].decode('utf-16-le')
def header(major,n):
 if n<24:return bytes([major*32+n])
 for width,ai in [(1,24),(2,25),(4,26),(8,27)]:
  if n<1<<(width*8):return bytes([major*32+ai])+n.to_bytes(width,'big')

def canonical(v):
 if v is None:return b'\xf6'
 if isinstance(v,bool):return b'\xf5' if v else b'\xf4'
 if isinstance(v,int):return header(0,v) if v>=0 else header(1,-1-v)
 if isinstance(v,float):return b'\xfb'+struct.pack('>d',v)
 if isinstance(v,str):
  b=v.encode('utf8',errors='strict');return header(3,len(b))+b
 if isinstance(v,bytes):return header(2,len(v))+v
 if isinstance(v,list):return header(4,len(v))+b''.join(map(canonical,v))
 if isinstance(v,dict):return header(5,len(v))+b''.join(canonical(k)+canonical(v[k]) for k in sorted(v,key=canonical))
 raise AuditError('Unsupported CBOR value')
def decode_cbor(data):

  if major==7:
   if ai in [20,21,22]:return {20:False,21:True,22:None}[ai]
   if ai==27:return struct.unpack('>d',read(8))[0]
   raise AuditError('Unsupported CBOR simple')
  need(ai<28,'Indefinite/reserved CBOR');n=ai if ai<24 else int.from_bytes(read({24:1,25:2,26:4,27:8}[ai]),'big')
  if major==0:return n
  if major==1:return -1-n
  if major==2:return read(n)
  if major==3:return read(n).decode('utf8',errors='strict')
  if major==4:return [item(depth+1) for _ in range(n)]
  if major==5:
   out={}
   for _ in range(n):
    k,v=item(depth+1),item(depth+1);need(isinstance(k,str) and k not in out,'Duplicate/nontext CBOR key');out[k]=v

  raise AuditError('Unsupported CBOR major')

def crc32c(data):
