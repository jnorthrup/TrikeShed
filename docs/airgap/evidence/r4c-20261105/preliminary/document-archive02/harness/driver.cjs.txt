const {chromium}=require('playwright');
const fs=require('fs'),path=require('path'),crypto=require('crypto'),assert=require('assert/strict');

let browser,context,page;
(async()=>{

 const waitFile=async name=>{const end=Date.now()+90000;while(!fs.existsSync(path.join(root,name))){assert(Date.now()<end,'Restart timed out');await new Promise(r=>setTimeout(r,100));}return JSON.parse(fs.readFileSync(path.join(root,name)));};

 context.on('request',r=>{if(r.method()==='POST')report.requests.push({url:r.url(),body:r.postData()});});

 const debug=await get('/api/lcnc/run/debug?token='+request.debugToken+'&runId='+old.runId);assert.equal(debug.state,'paused');assert.equal(debug.site.nodeId,stage);
 report.running=old;report.liveDebug=debug;fs.writeFileSync(path.join(root,'live-debug-before-crash.json'),JSON.stringify(debug,null,2));
 report.cases.ownedBoundary={result:'PASS',before:stage,runId:old.runId,ordinal:debug.ordinal,observations:debug.observationSummary};
 await page.screenshot({path:path.join(root,'before-crash.png'),fullPage:true});save();
 fs.writeFileSync(path.join(root,'restart-request.json'),JSON.stringify({boot:seed.boot,runId:old.runId,before:stage}));
 const restart=await waitFile('restart-ready.json');assert.equal(restart.firstExit,137);assert.equal(restart.javaReaped,true);assert.equal(restart.cleanupRan,false);

 const expired=await context.request.get(base+'/api/lcnc/run/debug?token='+request.debugToken+'&runId='+old.runId);assert.equal(expired.status(),404);const expiredBody=await expired.json();assert.equal(expiredBody.error,'debug_run_not_active');assert.equal(expiredBody.receipt.status,'interrupted');assert.equal(expiredBody.receipt.runId,old.runId);

 const ledger=await get('/api/lcnc/content?cid='+encodeURIComponent(completed.invocationsCid));assert.deepEqual(ledger.observations.filter(e=>e.phase==='consumed').map(e=>e.wire.index),[0,1,2,3,4,5,6,7]);

 report.cases.explicitNewRun={result:'PASS',oldRunId:old.runId,newRunId:completed.runId,programCid:completed.programCid,oldReceiptUnchanged:true,sourceRevisionsUnchanged:true,consumedWires:8,quote:quote.quote,nativeLiveBytesAfter:0};
 await page.screenshot({path:path.join(root,'explicit-new-run.png'),fullPage:true});save();await closePage('recoveryVideo');
 fs.writeFileSync(path.join(root,'inspection-request.json'),JSON.stringify({boot:recovered.boot,receiptCid:completed.receiptCid}));const marker=await waitFile('inspection-ready.json');assert.equal(marker.secondExit,0);
 const inspected=await get('/__proof/state');assert.equal(inspected.mode,'inspect');assert.equal(inspected.llvmInstalled,false);assert.equal(inspected.minerInstalled,false);assert.equal(inspected.assetsVisible,false);assert.equal(inspected.modulesVisible,false);assert(![seed.boot,recovered.boot].includes(inspected.boot));assert.deepEqual(inspected.runs,now.runs);assert.deepEqual(inspected.documents,seed.documents);
 await open();d=await receipt(completed.receiptCid);const show=d.getByRole('button',{name:'Show recorded outputs on matching canvas',exact:true});await show.click();await d.getByRole('button',{name:'Close facets',exact:true}).click();

 report.cases.freshInspector={result:'PASS',thirdBoot:inspected.boot,identicalRows:true,identicalDocuments:true,noNativeOrMiner:true,postRuns:2};await page.screenshot({path:path.join(root,'fresh-inspection.png'),fullPage:true});report.inspectionVideo=path.basename(await page.video().path());
 assert.equal(report.errors.length,0);save();await context.request.post(base+'/__proof/finish');
})().catch(async e=>{report.failure=String(e.stack||e);save();if(page)await page.screenshot({path:path.join(root,'failure.png'),fullPage:true}).catch(()=>{});console.error(e);process.exitCode=1;}).finally(async()=>{if(context)await context.close().catch(()=>{});if(browser)await browser.close().catch(()=>{});save();});
