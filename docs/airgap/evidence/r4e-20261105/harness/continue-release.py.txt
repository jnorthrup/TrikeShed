import subprocess,os
h=Path('/workspace/airgap-r4e-proof')

 with (Path('/workspace')/('r4e-'+name+'.txt')).open('w') as log:p=subprocess.run(args,stdout=log,stderr=subprocess.STDOUT,env=env)
 assert p.returncode==0,(name,p.returncode)
 print(name,'PASS',flush=True)
env=os.environ.copy();env['TRIKESHED_PROOF_ROOT']=str(h/'unpacked/airgap-document-rootfs')
prefix=['unshare','--user','--map-root-user','--net','sh','-c','ip link set lo up && exec "$@"','r4e-proof','python']
run(['python',str(h/'package.py'),'verify'],'unpacked-reverify',env)
