import subprocess,json,datetime

def git(*args):return subprocess.check_output(['git',*args],cwd=repo,text=True)
assert not git('status','--porcelain')
heads={};fetch={}
for remote in git('remote').splitlines():
 heads[remote]=git('ls-remote','--heads',remote)
 p=subprocess.run(['git','fetch','--no-prune',remote,'refs/heads/*:refs/remotes/'+remote+'/*'],cwd=repo,text=True,stdout=subprocess.PIPE,stderr=subprocess.STDOUT);fetch[remote]={'exit':p.returncode,'output':p.stdout};assert p.returncode==0,p.stdout
refs=[]
