const {chromium}=require('playwright');const fs=require('fs'),path=require('path'),assert=require('assert/strict'),crypto=require('crypto');
const out=process.argv[2],base=process.argv[3];let browser,context,page;const report={requests:[],errors:[],cases:{}};
(async()=>{
 browser=await chromium.launch({executablePath:'/usr/bin/chromium',headless:true,args:['--no-sandbox','--disable-dev-shm-usage']});context=await browser.newContext({viewport:{width:1800,height:1200},recordVideo:{dir:path.join(out,'videos'),size:{width:1800,height:1200}}});page=await context.newPage();page.on('pageerror',e=>report.errors.push(String(e)));context.on('request',r=>{if(r.method()==='POST')report.requests.push({url:r.url(),body:r.postData()});});
 await page.goto(base+'/panels?load=airgap-document');await page.waitForFunction(()=>document.querySelectorAll('.node').length===5&&document.body.innerText.includes('Loaded for review · choose Run or Live to execute'));await page.waitForFunction(()=>location.hash.length>1);await page.locator('#panelName').fill('airgap-coinbase-suski');
 const response=page.waitForResponse(r=>new URL(r.url()).pathname==='/api/lcnc/run'&&r.request().method()==='POST',{timeout:120000});await page.locator('#debugBtn').click();const pause=async id=>page.waitForFunction(id=>document.querySelector('[data-debug-state]')?.textContent.includes('Paused before '+id),id,{timeout:60000});await pause('document');await page.getByRole('combobox',{name:'Activity display speed'}).selectOption('0.25');
 for(const id of ['source','passages','scan']){await page.getByRole('button',{name:'Step into next node',exact:true}).click();await pause(id);}
 await page.waitForSelector('[data-observed-wire]');await page.waitForTimeout(450);const positions=()=>page.locator('[data-observed-wire]').evaluateAll(es=>es.map(e=>({wire:e.getAttribute('data-observed-wire'),x:e.getAttribute('cx'),y:e.getAttribute('cy')})));
 const before=await positions();assert(before.length);await page.getByRole('combobox',{name:'Activity display speed'}).selectOption('4');await page.waitForTimeout(40);const after=await positions();assert.equal(after.length,0,'Baseline speed increase discards unfinished animation');
 report.cases.speedDiscontinuity={result:'REPRODUCED',before,after};report.cases.missingInspection={result:'REPRODUCED',pauseDisplayButtons:await page.getByRole('button',{name:'Pause display',exact:true}).count(),tables:await page.getByRole('table',{name:'Observed transfer counts'}).count()};assert.equal(report.cases.missingInspection.pauseDisplayButtons,0);assert.equal(report.cases.missingInspection.tables,0);
 await page.getByRole('button',{name:'Continue run',exact:true}).click();report.receipt=await(await response).json();assert.equal(report.receipt.status,'completed');assert.equal(report.receipt.outputs.scan.runtime.liveBytesAfter,0);assert.equal(report.errors.length,0);await page.screenshot({path:path.join(out,'baseline.png'),fullPage:true});report.video=path.basename(await page.video().path());
})().catch(async e=>{report.error=String(e.stack||e);if(page)await page.screenshot({path:path.join(out,'failure.png'),fullPage:true}).catch(()=>{});process.exitCode=1;}).finally(async()=>{if(context)await context.close();if(browser)await browser.close();fs.writeFileSync(path.join(out,'browser.json'),JSON.stringify(report,null,2));});
