class AuditError(Exception):
    pass

def need(condition, message):
    if not condition:
        raise AuditError(message)

def digest(data):
    return hashlib.sha256(data).hexdigest()

def cid(data):
    return 'sha256:' + digest(data)

def strict_json(data):
    def pairs(rows):
        result = {}
        for k, v in rows:
            need(k not in result, 'duplicate JSON key: ' + k)
            result[k] = v
        return result
    return json.loads(data, object_pairs_hook=pairs,
                      parse_constant=lambda value: (_ for _ in ()).throw(AuditError('nonfinite JSON number')))

def load(path):
    return strict_json(path.read_bytes().decode('utf-8', errors='strict'))

def cbor_head(major, value):
    if value < 24:
        return bytes([(major << 5) | value])
    for width, ai in [(1, 24), (2, 25), (4, 26), (8, 27)]:
        if value < 1 << (8 * width):
            return bytes([(major << 5) | ai]) + value.to_bytes(width, 'big')
    raise AuditError('CBOR integer exceeds 64 bits')

def cbor_encode(value):
    if value is None:
        return b'\xf6'
    if isinstance(value, bool):
        return b'\xf5' if value else b'\xf4'
    if isinstance(value, int):
        return cbor_head(0, value) if value >= 0 else cbor_head(1, -1 - value)
    if isinstance(value, str):
        b = value.encode('utf-8', errors='strict')
        return cbor_head(3, len(b)) + b
    if isinstance(value, bytes):
        return cbor_head(2, len(value)) + value
    if isinstance(value, list):
        return cbor_head(4, len(value)) + b''.join(cbor_encode(v) for v in value)
    if isinstance(value, dict):
        need(all(isinstance(k, str) for k in value), 'CBOR map keys must be strings')
        return cbor_head(5, len(value)) + b''.join(cbor_encode(k) + cbor_encode(value[k]) for k in sorted(value, key=cbor_encode))
    if isinstance(value, float):
        return b'\xfb' + struct.pack('>d', value)
    raise AuditError('unsupported CBOR value')

def cbor_decode(data):
    at = 0
    def read(n):
        nonlocal at
        need(at + n <= len(data), 'truncated CBOR')
        out = data[at:at+n]
        at += n
        return out
    def item(depth=0):
        need(depth <= 64, 'CBOR depth')
        tag = read(1)[0]
        major, ai = tag >> 5, tag & 31
        if major == 7:
            if ai in [20, 21, 22]:
                return {20: False, 21: True, 22: None}[ai]
            if ai == 27:
                return struct.unpack('>d', read(8))[0]
            raise AuditError('unsupported CBOR simple/float')
        need(ai < 28, 'indefinite/reserved CBOR length')
        size = ai if ai < 24 else int.from_bytes(read({24: 1, 25: 2, 26: 4, 27: 8}[ai]), 'big')
        if major == 0:
            return size
        if major == 1:
            return -1 - size
        if major == 2:
            return read(size)
        if major == 3:
            return read(size).decode('utf-8', errors='strict')
        if major == 4:
            return [item(depth+1) for _ in range(size)]
        if major == 5:
            out = {}
            for _ in range(size):
                k, v = item(depth+1), item(depth+1)
                need(isinstance(k, str) and k not in out, 'CBOR duplicate/nontext map key')
                out[k] = v
            return out
        raise AuditError('unsupported CBOR major type')
    out = item()
    need(at == len(data), 'CBOR trailing bytes')
    need(cbor_encode(out) == data, 'CBOR differs from recursive encoded-key canonical encoding')
    return out

def zero_counts(row, label, runs=0):
    for key in ['freshInspectionPosts', 'modelCalls', 'automaticIntake', 'reviewIntakeCalls', 'analysisReteFacts']:
        need(row[key] == 0, label + ': unexpected ' + key)
    need(row['authoredRunPosts'] == runs and row['queryCalls'] == runs, label + ': run/query count')
    need(row['configuredModelCount'] == 0 and row['guestModules'] == {'installed': [], 'mounted': []}, label + ': configured modules/providers')
    need(row['blackboardUnchangedFromReadinessBaseline'] is True, label + ': analysis blackboard changed')

def validate_query(run, inspection, source_text):
    need(run['ok'] is True and run['status'] == 'completed', 'run did not complete')
    q = run['outputs']['query']
    need(q['dataset'] == 'english-receipt-observations', 'wrong query dataset')
    src = q['source']
    expected = {'receiptCid': INSPECTION, 'originalCid': SOURCE, 'extractedTextCid': SOURCE,
                'mode': 'current', 'currentness': 'checked-before-and-after-query',
                'historical': False, 'reviewStatus': 'unreviewed',
                'semanticAuthority': 'not-established', 'producerAuthority': 'not-established',
                'parserExecution': 'not-reverified', 'namedGraph': 'urn:trikeshed:english:' + INSPECTION[7:]}
    for k, v in expected.items():
        need(src[k] == v, 'query source ' + k)
    rows = q['results']['results']['bindings']
    need(len(rows) == 2, 'query must return exactly two syntax rows')
    need(sorted((r['subjectWord']['value'], r['predicateWord']['value']) for r in rows) ==
         [('Person', 'Agent'), ('Tenant', 'Person')], 'source-cased syntax words')
    for row in rows:
        need('quote' not in row, 'word result falsely labelled quotation')
        nodes, sentence_id = {}, None
        for key in ['head', 'subject', 'copula']:
            term = row[key]
            need(term['type'] == 'uri', 'query node is not a receipt URI')
            prefix = src['namedGraph'] + '/sentence/'
            need(term['value'].startswith(prefix), 'node belongs to another receipt')
            match = re.fullmatch(r'(\d+)/node/(.+)', term['value'][len(prefix):])
            need(match is not None, 'query node URI shape')
            sid, node_id = int(match[1]), match[2]
            need(sentence_id in [None, sid], 'binding crosses source sentences')
            sentence_id = sid
            sentence = next((s for s in inspection['sentences'] if s['index'] == sid), None)
            need(sentence is not None, 'unknown source sentence')
            nodes[key] = next((n for n in sentence['nodes'] if n['id'] == node_id), None)
            need(nodes[key] is not None, 'unknown full parser node identity')
        need(sentence['quote'] == source_text[sentence['begin']:sentence['end']], 'source quotation mismatch')
        need(nodes['head']['word'] == row['predicateWord']['value'] and
             nodes['subject']['word'] == row['subjectWord']['value'], 'binding differs from source node words')
        for relation, target in [('cop', 'copula'), ('nsubj', 'subject')]:
            need(any(e['relation'] == relation and e['governorNode'] == nodes['head']['id'] and
                     e['dependentNode'] == nodes[target]['id'] for e in sentence['dependencies']), 'binding lacks recorded parser edge')
    need(run['outputs']['bindings']['x'] == q['results'] and run['outputs']['source']['x'] == src, 'display outputs mismatch query')

def validate_ledger(inv, run, program):
    need(inv['kind'] == 'lcnc-invocations' and inv['runId'] == run['runId'] and
         inv['programCid'] == run['programCid'], 'ledger run/program identity')
    need(inv['observationSchema'] == 'lcnc-observations/1' and 0 < len(inv['observations']) <= 128, 'ledger observations absent/oversized')
    previous, consumed = 0, 0
    entries = {e['ordinal']: e for e in inv['entries']}
    for e in inv['observations']:
        need(isinstance(e['sequence'], int) and e['sequence'] > previous, 'observation ordering')
        previous = e['sequence']
        if e['phase'] != 'consumed' or not e.get('wire'):
            continue
        consumed += 1
        need(e['documentCid'] == run['programCid'] and e['status'] == 'delivered' and e['origin'] == 'wire', 'consumed observation identity')
        need(e['evidence'] == {'status': 'unavailable', 'reason': 'value_identity_not_recorded'}, 'consumed metadata falsely claims value identity')
        w = e['wire']
        need(0 <= w['index'] < len(program['wires']), 'consumed wire index')
        actual = program['wires'][w['index']]
        need(actual['from'] == [w['fromNode'], w['fromPort']] and actual['to'] == [w['toNode'], w['toPort']], 'consumed wire differs from program')
        producer = e['producer']
        need(producer['documentCid'] == run['programCid'] and producer['nodeId'] == w['fromNode'] and e['nodeId'] == w['toNode'], 'producer/consumer frame ownership')
        need(entries[e['invocationOrdinal']]['nodeId'] == e['nodeId'] and
             entries[producer['invocationOrdinal']]['nodeId'] == producer['nodeId'], 'observation invocation binding')
    need(consumed > 0, 'no actual consumed wire observations')

def mutation_controls(browser):
    mutations = {
        'wrong-inspection-I': lambda b: b['observed']['runs']['B']['run']['outputs']['query']['source'].__setitem__('receiptCid', SOURCE),
        'wrong-source-quotation': lambda b: b['observed']['inspection']['sentences'][0].__setitem__('quote', 'Invented source quotation'),
        'wrong-program-CID': lambda b: b['observed']['runs']['A_success']['run'].__setitem__('programCid', SOURCE),
        'wrong-run-ID': lambda b: b['observed']['runs']['A_success']['invocations'].__setitem__('runId', b['observed']['runs']['B']['run']['runId']),
        'wrong-activity-owner': lambda b: b['holds'][0]['after'].__setitem__('banner', 'Recorded playback · no execution ' + b['observed']['runs']['A_success']['run']['runId']),
        'automatic-run-before-explicit-Run': lambda b: b['observed']['negativeGuards']['counters'].__setitem__('authoredRunPosts', 1),
        'false-consumed-value-authority': lambda b: next(e for e in b['observed']['runs']['B']['invocations']['observations'] if e['phase'] == 'consumed' and e.get('wire'))['evidence'].__setitem__('status', 'verified'),
        'automatic-run-on-Import-Fit': lambda b: b['observed']['handoffRestorations'][0].__setitem__('automaticRuns', 1),
    }
    results = []
    for name, mutate in mutations.items():
        value = copy.deepcopy(browser)
        mutate(value)
        try:
            validate_browser(value)
        except AuditError as error:
            results.append({'mutation': name, 'result': 'REJECTED', 'reason': str(error)})
        else:
            raise AuditError('mutated-copy control accepted: ' + name)
    return results

def audit(native, snapshot, archive):
    checks, artifacts = [], {}
    def checked(name):
        checks.append(name)
    def file_bytes(path):
        data = path.read_bytes()
        artifacts[str(path)] = {'bytes': len(data), 'sha256': digest(data)}
        return data
    missing = [str(native / f) for f in REQUIRED if not (native / f).is_file()]
    if missing:
        return {'result': 'PENDING', 'missing': missing, 'checks': [], 'mutatedCopyControls': []}
    b, e, cleanup, reopened, startup, expected, setup, before = [load(native / f) for f in
        ['browser-result.json', 'fixture-evidence.json', 'cleanup.json', 'native-reopened-cas.json',
         'startup-evidence.json', 'expected-inspection.json', 'setup.json', 'archive-manifest-before.json']]
    for name, sha in FROZEN.items():
        need(digest(file_bytes(snapshot / name)) == sha, 'frozen artifact changed: ' + name)
    need((snapshot / 'source-head.txt').read_text().strip() == SOURCE_HEAD, 'frozen source head')
    declared = load(snapshot / 'sha256.json')
    for name, sha in declared.items():
        if name == 'dependencies':
            continue
        need(digest(file_bytes(snapshot / name)) == sha, 'snapshot hash manifest: ' + name)
    checked('Frozen source, main JAR, three JS artifacts and static snapshot hash manifest')
    validate_browser(b)
    checked('Exact eight-field action, inert offered current construction/context, ten labelled native negatives, three actual same-program runs, two held A/B ownership controls')
    need(cid(file_bytes(archive)) == 'sha256:' + ARCHIVE_SHA, 'original archive SHA256')
    need(before['archiveSha256'] == ARCHIVE_SHA and setup['sourceArchiveSha256'] == ARCHIVE_SHA, 'archive provenance manifest')
    with tarfile.open(archive) as tar:
        archived = {m.name: tar.extractfile(m).read() for m in tar.getmembers() if m.isfile()}
    for row in before['originalFiles']:
        data = archived[row['path']]
        need(len(data) == row['bytes'] and digest(data) == row['sha256'], 'original archive file manifest')
    need(len(archived) == 7, 'original archive file count')
    project, document = setup['selection']['project'], setup['selection']['id']
    protected = {'project-couch/' + project + '/HEAD', 'project-manifests/' + project + '.tsv',
                 'project-intake-policy/' + project + '.guard', 'files/' + project + '/' + document}
    need({row['path'] for row in e['originalArchiveHeadsAndSource']} == protected and
         len(e['originalArchiveHeadsAndSource']) == 4, 'four original source/head/manifest/guard bindings required')
    for row in e['originalArchiveHeadsAndSource']:
        relative = 'runtime/' + row['path']
        data = file_bytes(native / relative)
        need(data == archived[relative] and row['unchanged'] is True and row['sha256'] == cid(data) and row['bytes'] == len(data), 'original source/head/manifest/guard changed')
    pack = file_bytes(native / 'runtime/cas/pack')
    need(pack.startswith(archived['runtime/cas/pack']), 'original CAS pack prefix changed')
    checked('Original seven-file archive identity; protected source/head/manifest/guard and CAS pack prefix preserved')
    blocks = {}
    for row in e['retainedReadbacks']:
        identity = row['cid']
        need(re.fullmatch(r'sha256:[0-9a-f]{64}', identity) is not None, 'invalid readback CID')
        data = file_bytes(native / 'cas-readbacks' / (identity[7:] + '.' + row['encoding']))
        need(cid(data) == identity and len(data) == row['bytes'], 'retained block identity/size')
        need(identity not in blocks, 'duplicate retained block')
        blocks[identity] = (row['encoding'], data)
    need(blocks[SOURCE] == ('txt', SOURCE_TEXT.encode()), 'original source block')
    need(blocks[INSPECTION][0] == 'cbor', 'original inspection encoding')
    original = cbor_decode(blocks[INSPECTION][1])
    need(blocks[INSPECTION][1] in archived['runtime/cas/pack'], 'original receipt absent from archived CAS pack')
    need({**original, 'receiptCid': INSPECTION} == expected, 'CBOR receipt differs from exported original inspection')
    need({k: v for k, v in b['observed']['inspection'].items() if k != 'retrieval'} == expected, 'opened inspection differs from original receipt')
    need(original['sourceSnapshots'] == e['retainedSourceSnapshots'] == startup['retainedSourceSnapshots'] == reopened['recordedSourceSnapshots'], 'source snapshot identity')
    need(e['sourceCurrent'] is True and startup['sourceCurrent'] is True and e['sourceRevision']['cid'] == SOURCE, 'source currentness records')
    for key, value in setup['selection'].items():
        need(b['meta']['sourceRevision'][key] == e['sourceRevision'][key] == startup['sourceRevision'][key] == value, 'source revision binding: ' + key)
    checked('Original 3f958 canonical CBOR round trip, source CID/quotations and retained source snapshots')
    expected_retained = set(blocks)
    for label, record in b['observed']['runs'].items():
        run = record['run']
        for field, key in [('receiptCid', 'durable'), ('programCid', 'program'), ('invocationsCid', 'invocations')]:
            identity = run[field]
            need(identity in blocks and blocks[identity][0] == 'json', 'run block absent from actual CAS exports')
            raw = file_bytes(native / 'durable-readbacks' / (identity[7:] + '.json'))
            need(raw == blocks[identity][1] and strict_json(raw.decode('utf-8')) == record[key], 'durable/CAS/browser block mismatch: ' + label + '/' + field)
    for hold in b['holds']:
        raw = file_bytes(native / 'held-responses' / (hold['cid'][7:] + '.json'))
        need(cid(raw) == hold['sha256'] == hold['cid'] and len(raw) == hold['bytes'] and
             raw == blocks[hold['cid']][1] and strict_json(raw.decode('utf-8')) == hold['actualResponse'], 'held A original bytes changed')
    imported = file_bytes(native / 'native-imports/actual-B-admitted-document.json')

def main():
    p = argparse.ArgumentParser(description=__doc__)
    p.add_argument('native', type=Path)
    p.add_argument('snapshot', type=Path)
    p.add_argument('--archive', type=Path, default=Path('/workspace/library-files/english-retained-opener-inputs/positive-unmutated-runtime.tar.gz'))
    p.add_argument('--out', type=Path, default=Path(__file__).resolve().parent)
    args = p.parse_args()
    args.out.mkdir(parents=True, exist_ok=True)
    try:
        result = audit(args.native.resolve(), args.snapshot.resolve(), args.archive.resolve())
    except (AuditError, KeyError, ValueError, OSError, TypeError, IndexError, StopIteration) as error:
        result = {'result': 'FAIL', 'error': str(error), 'nativeDirectory': str(args.native.resolve())}
    result['auditScriptSha256'] = digest(Path(__file__).read_bytes())
    (args.out / 'audit-result.json').write_text(json.dumps(result, indent=2, ensure_ascii=False) + '\n')
    lines = ['Independent recorded-data audit: ' + result['result'], '',
             'Selected records: ' + str(args.native.resolve()), '']
    lines += [('- ' + item) for item in result.get('checks', [])]
    if result.get('missing'):
        lines += ['', 'Required final records are absent; no final PASS is issued.'] + result['missing']
    if result.get('error'):
        lines += ['', result['error']]
    lines += ['', 'This audits recorded evidence only. It does not claim semantic entailment, producer authentication, fresh parser validation or a global airgap.']
    (args.out / 'report.txt').write_text('\n'.join(lines) + '\n')
    print(json.dumps({'result': result['result'], 'checks': len(result.get('checks', [])),
                      'mutatedCopyControls': len(result.get('mutatedCopyControls', [])), 'report': str(args.out / 'audit-result.json')}))
    return {'PASS': 0, 'PENDING': 2, 'FAIL': 1}[result['result']]

if __name__ == '__main__':
    sys.exit(main())
