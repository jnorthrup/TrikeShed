const fs=require('node:fs'),path=require('node:path'),assert=require('node:assert/strict');
const {chromium}=require('playwright');
async function main(){
 const target=new URL(process.argv[2]);assert.equal(target.protocol,'http:');assert.equal(target.hostname,'127.0.0.1');
 const base=target.origin,art=__dirname,errors=[],writes=[],external=[];let browser,page;
 const control=async(name,method='GET')=>{const r=await fetch(base+'/_fixture/'+name,{method});assert.equal(r.status,200,name);return r.json();};
 const content=async cid=>{const r=await fetch(base+'/api/lcnc/content?cid='+encodeURIComponent(cid));assert.equal(r.status,200);return r.json();};
 const original=JSON.parse(fs.readFileSync('/workspace/library-files/documents-legal-candidates-proof/result.json','utf8'));
 const authored=()=>writes.filter(w=>w.path==='/api/lcnc/run'&&w.body.document);
 const rows=query=>query.results.results.bindings;
 const binding=(row,key)=>row[key].value;
 try{
  const meta=await control('meta'),programs=meta.panels.programs,runs={},retained={},innerGraphs={};
  browser=await chromium.launch({headless:true,executablePath:'/usr/bin/chromium',args:['--no-sandbox']});page=await browser.newPage({viewport:{width:1920,height:1400},reducedMotion:'reduce'});
  page.on('pageerror',e=>errors.push(e.message));page.on('request',r=>{const u=new URL(r.url());if(u.origin!==base)external.push(r.url());else if(r.method()==='POST')writes.push({path:u.pathname,body:r.postDataJSON()});});
  const load=async name=>{const before=authored().length;await page.goto(base+'/panels?load='+name,{waitUntil:'domcontentloaded'});await page.locator('.node[data-id="query"],.node[data-id="select"]').first().waitFor({timeout:45000});assert.equal(await page.locator('.node').count(),programs[name].nodes.length);assert.equal(authored().length,before,'loading does not run the authored graph');const palette=page.locator('#palette');if((await palette.getAttribute('class')||'').includes('open'))await page.locator('#paletteBtn').click();await page.locator('#fitBtn').click();};
  for(const name of ['legal-inner-thirty-days','legal-inner-unknown-quantity']){await load(name);assert.equal(await page.locator('#runBtn').isDisabled(),true);assert.equal(programs[name].controls.inspectionOnly,true);assert(programs[name].nodes.some(n=>n.type==='sparql.pattern'&&n.params.predicate==='<urn:trikeshed:english:legal:kind>'&&n.params.object==='"duration-language-candidate"'));innerGraphs[name]=programs[name];await page.screenshot({path:path.join(art,name+'.png'),fullPage:true});}
  const checkQuery=(query,inspection,candidate,mode)=>{assert.equal(query.dataset,'english-legal-candidate-observations');const s=query.source;assert.equal(s.inspectionCid,inspection);assert.equal(s.candidateCid,candidate);assert.equal(s.receiptCid,candidate);assert.equal(s.mode,mode);assert.equal(s.historical,mode==='snapshot');assert.equal(s.namedGraph,'urn:trikeshed:english:legal:'+candidate.slice(7));assert.equal(s.reviewStatus,'unreviewed');assert.equal(s.parserExecution,'not-reverified');assert.equal(s.producerAuthority,'not-established');assert.equal(s.sameSentenceJoin,'cooccurrence-only-not-applicability');assert.equal(s.integerValueMeaning,'bounded-lexical-reading-only');assert.equal(s.missingIntegerValue,'unresolved-or-unrepresented-not-false');assert.equal(s.currentness,mode==='current'?'checked-before-and-after-query':'not-checked');};
  const execute=async(key,name,expected=true)=>{await load(name);assert(programs[name].nodes.some(n=>n.type==='document.english.patterns.query'));assert.equal(programs[name].wires.length,8);assert.equal(programs[name].nodes.find(n=>n.id==='pattern').type,'json.value');const syntax=JSON.parse(programs[name].nodes.find(n=>n.id==='pattern').params.value);assert.equal(syntax.controls.inspectionOnly,true);if(key==='matching')await page.screenshot({path:path.join(art,'legal-query-connected-inert-canvas.png'),fullPage:true});
   const waiting=page.waitForResponse(r=>new URL(r.url()).pathname==='/api/lcnc/run'&&r.request().method()==='POST'&&r.request().postDataJSON().document,{timeout:90000});await page.locator('#runBtn').click();const response=await waiting;const run=await response.json();runs[key]=run;fs.writeFileSync(path.join(art,'progress.json'),JSON.stringify({meta,runs,retained,innerGraphs,writes,errors,external},null,2));assert.equal(response.status(),expected?200:400);assert.equal(run.ok,expected,JSON.stringify(run));assert.equal(run.status,expected?'completed':'failed');assert.match(run.receiptCid,/^sha256:[a-f0-9]{64}$/);const durable=await content(run.receiptCid);retained[key]=durable;assert.equal(durable.lcncRun.runId,run.runId);assert.equal(durable.lcncRun.programCid,run.programCid);assert.equal(durable.lcncRun.status,run.status);fs.writeFileSync(path.join(art,'progress.json'),JSON.stringify({meta,runs,retained,innerGraphs,writes,errors,external},null,2));await page.waitForFunction(status=>document.getElementById('status').textContent.includes('Program '+status),run.status,{timeout:30000});
   if(expected){assert.deepEqual(run.outputs.query,durable.lcncRun.outputs.query);assert.deepEqual(run.outputs.bindings.x,run.outputs.query.results);assert.deepEqual(run.outputs.source.x,run.outputs.query.source);const drawer=page.getByRole('region',{name:'Workspace facets'});await drawer.waitFor({timeout:30000});await drawer.getByRole('button',{name:'Refresh',exact:true}).click();await drawer.getByRole('button',{name:'Inspect receipt',exact:true}).click();await drawer.getByText(/Recorded node outputs/).waitFor({timeout:30000});await drawer.locator('summary').filter({hasText:/^query$/}).click();await drawer.getByText('Receipt: '+run.receiptCid,{exact:false}).scrollIntoViewIfNeeded();await page.screenshot({path:path.join(art,'legal-query-'+key+'-durable-evidence.png'),fullPage:true});if(key==='matching'||key==='unknownB'){for(const id of ['bindings','source']){const summary=drawer.locator('summary').filter({hasText:new RegExp('^'+id+'$')});await summary.click();const detail=summary.locator('..');await detail.scrollIntoViewIfNeeded();const pre=detail.locator('pre');if(id==='bindings')await pre.evaluate(el=>{el.scrollTop=150;});await detail.screenshot({path:path.join(art,'legal-query-'+key+'-'+id+'-detail.png')});}}await page.getByRole('button',{name:'Close facets',exact:true}).click();await page.locator('.node[data-id="query"] .result').waitFor({timeout:15000});await page.locator('#fitBtn').click();await page.screenshot({path:path.join(art,'legal-query-'+key+'-completed-canvas.png'),fullPage:true});}
   else{assert.match(JSON.stringify(run.error||durable.lcncRun.error||{}),/stale|changed|source snapshot|revision/i);await page.screenshot({path:path.join(art,'legal-query-current-stale-failed-receipt.png'),fullPage:true});}
   return run;
  };
  const matching=await execute('matching','legal-known-current-a');checkQuery(matching.outputs.query,original.inspectionA.receiptCid,original.currentA.receiptCid,'current');assert.equal(rows(matching.outputs.query).length,1);let row=rows(matching.outputs.query)[0];assert.equal(binding(row,'quote'),'within thirty (30) business days');assert.equal(binding(row,'sentence'),original.currentA.sentences[0].quote);assert.equal(binding(row,'source'),'urn:sha256:'+original.currentA.extractedTextCid.slice(7));assert.equal(matching.outputs.query.source.candidateCid,original.currentA.receiptCid);assert.equal(matching.outputs.query.source.sameSentenceJoin,'cooccurrence-only-not-applicability');
  const knownB=await execute('knownB','legal-known-current-b');checkQuery(knownB.outputs.query,original.inspectionB.receiptCid,original.currentB.receiptCid,'current');assert.equal(rows(knownB.outputs.query).length,1);row=rows(knownB.outputs.query)[0];assert.equal(binding(row,'quote'),'within 30 calendar days');assert.equal(binding(row,'sentence'),original.currentB.sentences[1].quote);assert(!rows(knownB.outputs.query).some(r=>binding(r,'quote').includes('thirty (60)')),'number-form30 on conflicting duration must not satisfy known integer30');
  const unknownB=await execute('unknownB','legal-unknown-current-b');checkQuery(unknownB.outputs.query,original.inspectionB.receiptCid,original.currentB.receiptCid,'current');assert(rows(unknownB.outputs.query).length>0);assert(rows(unknownB.outputs.query).every(r=>binding(r,'quote')==='within thirty (60) business days'));assert(rows(unknownB.outputs.query).some(r=>binding(r,'reason')==='conflicting-number-forms'));
  const mutation=await control('mutate','POST');await execute('stale','legal-known-current-a',false);const snapshot=await execute('snapshot','legal-known-snapshot-a');checkQuery(snapshot.outputs.query,original.inspectionA.receiptCid,original.historicalA.receiptCid,'snapshot');assert.equal(rows(snapshot.outputs.query)[0].sentence.value,original.currentA.sentences[0].quote);
  const evidence=await control('evidence');assert.equal(authored().length,5);assert.deepEqual(errors,[]);assert.deepEqual(external,[]);
  const result={result:'PASS',scope:'Actual immutable Panels connected outer graphs and inspection-only inner BGP canvases; explicit runs use actual KanbanModule durable inline service borrowing the actual candidate query owner. Reuses actual retained v58 CoreNLP4.6 observations, with no new parser. Source-linked unreviewed candidates, not legal applicability; unknown integer is not false.',meta,runs,retained,innerGraphs,mutation,evidence,writes,browserErrors:errors,externalBrowserRequests:external};fs.writeFileSync(path.join(art,'result.json'),JSON.stringify(result,null,2));console.log(JSON.stringify({result:'PASS',authoredRuns:5,matching:matching.receiptCid,unknownRows:rows(unknownB.outputs.query).length,snapshot:snapshot.receiptCid}));
 }catch(error){fs.writeFileSync(path.join(art,'failure-observations.json'),JSON.stringify({error:String(error),errors,writes,external},null,2));if(page)await page.screenshot({path:path.join(art,'failure.png'),fullPage:true}).catch(()=>{});throw error;}
 finally{if(browser)await browser.close();await control('finish','POST').catch(()=>{});}
}
main().catch(error=>{console.error(error.stack||error);process.exitCode=1;});
