package proof.legalpanels

import borg.trikeshed.lcnc.*
import borg.trikeshed.lib.*

/** External fixture data only: actual Panels edits these programs and the real runner executes them. */
object PanelPrograms {
    const val A = "legal-known-current-a"
    const val B = "legal-known-current-b"
    const val UNKNOWN = "legal-unknown-current-b"
    const val SNAPSHOT = "legal-known-snapshot-a"
    const val INNER = "legal-inner-thirty-days"
    const val INNER_UNKNOWN = "legal-inner-unknown-quantity"
    private fun predicate(name: String) = "<urn:trikeshed:english:legal:$name>"

    fun inner(unknown: Boolean = false): LcncProgram {
        val variables = if (unknown) s_["?quote", "?reason", "?sentence", "?candidate"]
            else s_["?quote", "?status", "?sentence", "?source"]
        val patterns = if (unknown) s_[
            Triple("?d", predicate("kind"), "\"duration-language-candidate\""),
            Triple("?d", predicate("status"), "\"unresolved-quantity-candidate\""),
            Triple("?d", predicate("quote"), "?quote"),
            Triple("?d", predicate("unresolved"), "?reason"),
            Triple("?d", predicate("sentence"), "?s"),
            Triple("?s", predicate("quote"), "?sentence"),
            Triple("?d", predicate("candidateReceipt"), "?candidate"),
        ] else s_[
            Triple("?d", predicate("kind"), "\"duration-language-candidate\""),
            Triple("?d", predicate("integerValue"), "\"30\"^^<http://www.w3.org/2001/XMLSchema#integer>"),
            Triple("?d", predicate("unitLemma"), "\"day\""),
            Triple("?d", predicate("quote"), "?quote"),
            Triple("?d", predicate("status"), "?status"),
            Triple("?d", predicate("sentence"), "?s"),
            Triple("?s", predicate("quote"), "?sentence"),
            Triple("?d", predicate("sourceText"), "?source"),
        ]
        val nodes = mutableListOf(LcncNode("select", SparqlGraph.SELECT, x = 1480.0, y = 380.0))
        val wires = mutableListOf<LcncWire>()
        for ((index, variable) in variables.view.withIndex()) {
            nodes.add(LcncNode("v$index", SparqlGraph.VARIABLE, _m["name" j variable, "ord" j index.toString()], x = 40.0, y = 50.0 + index * 180))
            wires.add(LcncWire("v$index", "variable", "select", "variables"))
        }
        for ((index, pattern) in patterns.view.withIndex()) {
            nodes.add(LcncNode("p$index", SparqlGraph.PATTERN, _m["subject" j pattern.first, "predicate" j pattern.second, "object" j pattern.third],
                x = 350.0 + (index % 3) * 360, y = 50.0 + (index / 3) * 270))
            wires.add(LcncWire("p$index", "pattern", "select", "patterns"))
        }
        return LcncProgram(if (unknown) INNER_UNKNOWN else INNER, nodes.toSeries(), wires.toSeries(),
            controls = LcncConfixControls(inspectionOnly = true), seq = nodes.size + 1)
    }

    fun outer(name: String, cid: String, mode: String, unknown: Boolean = false): LcncProgram = LcncProgram(name, s_[
        LcncNode("receipt", LcncContracts.SCOPE_IN, _m["name" j "inspectionCid", "kind" j "id", "default" j cid], x = 40.0, y = 120.0),
        LcncNode("mode", "text.value", _m["value" j mode], x = 40.0, y = 350.0),
        LcncNode("pattern", "json.value", _m["value" j LcncProgramConfix.toJson(inner(unknown))], x = 350.0, y = 120.0),
        LcncNode("query", EnglishLegalQueryContract.TYPE, x = 700.0, y = 120.0),
        LcncNode("bindings", "display", x = 1100.0, y = 100.0),
        LcncNode("source", "display", x = 1100.0, y = 370.0),
        LcncNode("dataset-out", LcncContracts.SCOPE_OUT, _m["name" j "dataset", "kind" j "text"], x = 1500.0, y = 100.0),
        LcncNode("results-out", LcncContracts.SCOPE_OUT, _m["name" j "results", "kind" j "json"], x = 1500.0, y = 330.0),
        LcncNode("source-out", LcncContracts.SCOPE_OUT, _m["name" j "source", "kind" j "json"], x = 1500.0, y = 570.0),
    ], s_[LcncWire("receipt", "value", "query", "inspectionCid"), LcncWire("mode", "value", "query", "mode"),
        LcncWire("pattern", "value", "query", "program"), LcncWire("query", "results", "bindings", "x"),
        LcncWire("query", "source", "source", "x"), LcncWire("query", "dataset", "dataset-out", "value"),
        LcncWire("query", "results", "results-out", "value"), LcncWire("query", "source", "source-out", "value")], seq = 10)

    fun programs(a: String, b: String): Map<String, LcncProgram> = _m[
        A j outer(A, a, "current"), B j outer(B, b, "current"), UNKNOWN j outer(UNKNOWN, b, "current", true),
        SNAPSHOT j outer(SNAPSHOT, a, "snapshot"), INNER j inner(), INNER_UNKNOWN j inner(true),
    ]
}
