const sha=b=>'sha256:'+crypto.createHash('sha256').update(b).digest('hex');
const decoderFile='/workspace/library-files/documents-legal-candidates-independent-audit/audit.cjs';
const decoderSource=fs.readFileSync(decoderFile,'utf8'),context={assert,TextDecoder,JSON,Buffer};
vm.createContext(context);vm.runInContext(decoderSource.slice(decoderSource.indexOf('function decode(bytes){'),decoderSource.indexOf('const checked=new Map()')),context);

 for(const sentence of receipt.sentences){assert.equal(sentence.quote,text.slice(sentence.begin,sentence.end));tokens+=sentence.tokens.length;nodes+=sentence.nodes.length;copies+=sentence.nodes.filter(n=>n.copy>0).length;for(const token of sentence.tokens)assert.equal(token.word,text.slice(token.begin,token.end));}
 selections.push({name,inspectionCid:entry.receiptCid,originalCid:receipt.originalCid,extractedTextCid:receipt.extractedTextCid,selection:receipt.selection,sourceSnapshotRows:entry.sourceSnapshots.length});
}

}
