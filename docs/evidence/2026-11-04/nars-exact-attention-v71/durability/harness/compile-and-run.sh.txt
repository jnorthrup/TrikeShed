#!/usr/bin/env bash
set -euo pipefail
source /workspace/.trikeshed-cloud/activate.sh
attention_proof=/workspace/library-files/nars-exact-attention-v71-proof/durability
attention_snapshot=/workspace/library-files/english-retained-attention-v71-proof/snapshot
attention_cp=$(cat "$attention_snapshot/classpath.txt")
attention_compiler=$(rg --files --no-ignore /workspace/.trikeshed-cloud/gradle/caches/modules-2/files-2.1/org.jetbrains.kotlin/kotlin-compiler-embeddable/2.4.21 | rg '\.jar$' | head -n 1)
mkdir -p "$attention_proof/classes"
java -Xmx768m -cp "$attention_compiler:$attention_cp" org.jetbrains.kotlin.cli.jvm.K2JVMCompiler \
 -no-stdlib -no-reflect -jvm-target 25 -classpath "$attention_cp" \
 -d "$attention_proof/classes" "$attention_proof/Proof.kt" > "$attention_proof/compile.log" 2>&1
java --enable-native-access=ALL-UNNAMED -Xmx768m -cp "$attention_proof/classes:$attention_cp" proof.nars.attention.ProofKt \
 writer "$attention_proof" > "$attention_proof/writer.log" 2>&1
java --enable-native-access=ALL-UNNAMED -Xmx768m -cp "$attention_proof/classes:$attention_cp" proof.nars.attention.ProofKt \
 reopen "$attention_proof" > "$attention_proof/reopen.log" 2>&1
