const fs=require('node:fs'),path=require('node:path'),assert=require('node:assert/strict'),crypto=require('node:crypto');
const {chromium}=require('playwright');
async function main(){
 const target=new URL(process.argv[2]);assert.equal(target.protocol,'http:');assert.equal(target.hostname,'127.0.0.1');
 const base=target.origin,art=__dirname,priorBytes=fs.readFileSync(path.join(art,'attempt1','browser-progress.json')),prior=JSON.parse(priorBytes),writes=[...prior.writes],errors=[...prior.errors],external=[...prior.external],observations={...prior.observations};let browser,page;
 const hash=b=>'sha256:'+crypto.createHash('sha256').update(b).digest('hex');

 const authored=()=>writes.filter(w=>w.path==='/api/lcnc/run'&&w.body?.document);
 const parses=()=>writes.filter(w=>w.path==='/api/documents/english');

 try{
  const meta=await control('meta'),source=meta.source;assert.equal(Buffer.byteLength(source.text,'utf8'),228);assert.equal(hash(Buffer.from(source.text,'utf8')),source.cid);assert.equal(fs.readFileSync(meta.nativeFilePath).length,228);assert.equal(hash(fs.readFileSync(meta.nativeFilePath)),source.cid);
  browser=await chromium.launch({headless:true,executablePath:'/usr/bin/chromium',args:['--no-sandbox']});page=await browser.newPage({viewport:{width:1920,height:1400},reducedMotion:'reduce'});
  page.on('pageerror',e=>errors.push(e.message));page.on('request',r=>{const u=new URL(r.url());if(u.origin!==base){external.push(r.url());return;}if(r.method()==='POST'&&!u.pathname.startsWith('/_fixture/')){let body;try{body=r.postDataJSON();}catch{body={binaryBytes:r.postDataBuffer()?.length};}writes.push({path:u.pathname,query:u.search,body});}});
  const inspection=observations.inspection;assert(inspection);assert.equal(inspection.originalCid,source.cid);assert.equal(inspection.selection.project,meta.project);assert.equal(inspection.selection.id,source.id);assert.equal(parses().length,1);assert.equal(authored().length,0);observations.continuation={scope:'Same retained fresh inspection and source rows from actual attempt1; no upload or parser call in continuation',priorProgressCid:hash(priorBytes),priorInspectionCid:inspection.receiptCid};persist();
  const cr=await fetch(base+'/api/panels/presets');assert.equal(cr.status,200);const catalog=await cr.json(),pattern=catalog.presets.find(p=>p.name==='preset-english-copular-pattern'),outer=catalog.presets.find(p=>p.name==='preset-english-copular-query');assert(pattern&&outer);assert.equal(pattern.document.controls.inspectionOnly,true);assert.equal(pattern.document.nodes.filter(n=>n.type==='sparql.pattern').length,8);assert.equal(pattern.document.nodes.length,14);assert.equal(outer.document.nodes.length,10);
  await page.goto(base+'/panels',{waitUntil:'domcontentloaded'});await page.locator('#presetsBtn').waitFor({timeout:45000});
  const loadPreset=async(title,expectedNodes)=>{const before=authored().length;await page.locator('#presetsBtn').click();const name=title==='Read a copular syntax diagram'?'preset-english-copular-pattern':'preset-english-copular-query',card=page.locator('#gallery .galRow[data-kind=\"preset\"][data-name=\"'+name+'\"]');await card.waitFor({timeout:30000});await card.click();await page.waitForFunction(n=>document.querySelectorAll('.node').length===n,expectedNodes,{timeout:30000});if((await page.locator('#palette').getAttribute('class')||'').includes('open'))await page.locator('#paletteBtn').click();await page.locator('#fitBtn').click();assert.equal(authored().length,before);assert.equal(parses().length,1);};
  await loadPreset('Read a copular syntax diagram',14);assert.equal(await page.locator('#runBtn').isDisabled(),true);assert.equal(await page.locator('#liveBtn').isDisabled(),true);assert.equal(await page.locator('#debugBtn').isDisabled(),true);assert.equal(await page.locator('.node[data-query-syntax="1"]').count(),14);assert.equal(await page.locator('.node .runtime-status').count(),0);assert.equal(await page.locator('.node .query-syntax-status').filter({hasText:'Query syntax · Inspection only'}).count(),14);assert.equal(authored().length,0);await page.screenshot({path:path.join(art,'native-copular-pattern-inert.png'),fullPage:true});observations.inertPattern={name:pattern.name,nodeCount:14,patternCount:8,runDisabled:true,liveDisabled:true,authoredRuns:0};persist();
  await loadPreset('Query copular syntax observations',10);const receipt=page.locator('.node[data-id="receipt"]');await receipt.getByRole('checkbox',{name:'Use default',exact:true}).check();await receipt.locator('.params label').filter({hasText:/^default/}).locator('xpath=following-sibling::input[1]').fill(inspection.receiptCid);await page.locator('.node[data-id="mode"] .params input:not([type=checkbox])').fill('current');assert.equal(await page.locator('#runBtn').isDisabled(),false);assert.equal(authored().length,0);
  const ran=page.waitForResponse(r=>new URL(r.url()).pathname==='/api/lcnc/run'&&r.request().method()==='POST'&&r.request().postDataJSON()?.document,{timeout:120000});await page.locator('#runBtn').click();const rr=await ran,run=await rr.json();observations.run=run;persist();assert.equal(rr.status(),200);assert.equal(run.status,'completed');assert.equal(run.ok,true);assert.equal(authored().length,1);assert.equal(parses().length,1);
  const query=run.outputs.query,bindings=query.results.results.bindings;assert.equal(query.dataset,'english-receipt-observations');assert.equal(bindings.length,3);const requirements=bindings.filter(r=>r.subjectWord.value==='requirements');assert.equal(requirements.length,2);assert.deepEqual(requirements.map(r=>r.subject.value.split('/').at(-1)).sort(),['2:0','2:3']);assert(requirements.every(r=>r.predicateWord.value==='clear'));const rent=bindings.filter(r=>r.subjectWord.value==='rent');assert.equal(rent.length,1);assert.equal(rent[0].predicateWord.value,'due');assert(rent[0].head.value.endsWith('/sentence/2/node/8:0'));assert(bindings.every(r=>!('quote' in r)));assert.equal(query.source.receiptCid,inspection.receiptCid);assert.equal(query.source.namedGraph,'urn:trikeshed:english:'+inspection.receiptCid.slice(7));assert.equal(query.source.originalCid,source.cid);assert.equal(query.source.mode,'current');assert.equal(query.source.historical,false);assert.equal(query.source.currentness,'checked-before-and-after-query');assert.equal(query.source.reviewStatus,'unreviewed');assert.equal(query.source.semanticAuthority,'not-established');assert.equal(query.source.producerAuthority,'not-established');assert.equal(query.source.parserExecution,'not-reverified');assert.deepEqual(run.outputs.bindings.x,query.results);assert.deepEqual(run.outputs.source.x,query.source);
  const durable=await readJson(run.receiptCid,'durable-readbacks');assert.deepEqual(durable.lcncRun.outputs,run.outputs);assert.equal(durable.lcncRun.runId,run.runId);const program=await readJson(run.programCid,'program-readbacks');const queryInput=program.nodes.find(n=>n.id==='receipt');assert.equal(queryInput.params.default,inspection.receiptCid);assert.equal(program.nodes.find(n=>n.id==='mode').params.value,'current');const invocations=await readJson(run.invocationsCid,'invocation-readbacks');assert.equal(invocations.runId,run.runId);assert.equal(invocations.programCid,run.programCid);const call=invocations.entries.find(e=>e.nodeId==='query');assert(call);assert.equal(call.status,'completed');if(call.inputsOmitted){assert.equal(typeof call.inputsOmitted,'string');observations.queryInvocationInputs={omitted:call.inputsOmitted,scope:'Bounded ledger does not retain the complete fourteen-node syntax program; identity/mode are verified from actual authored program and actual query source output.'};}else{assert.equal(call.inputs.receiptCid,inspection.receiptCid);assert.equal(call.inputs.mode,'current');}observations.durable=durable;observations.invocations=invocations;observations.program=program;observations.query=query;persist();
  await page.waitForFunction(()=>document.getElementById('status').textContent.includes('Program completed'),null,{timeout:30000});await page.locator('#fitBtn').click();await page.screenshot({path:path.join(art,'native-copular-query-completed.png'),fullPage:true});const drawer=page.getByRole('region',{name:'Workspace facets'});await drawer.waitFor();await drawer.getByRole('button',{name:'Refresh',exact:true}).click();await drawer.getByRole('button',{name:'Inspect receipt',exact:true}).click();await drawer.getByText(/Recorded node outputs/).waitFor();await drawer.locator('summary').filter({hasText:/^bindings$/}).click();const detail=drawer.locator('summary').filter({hasText:/^bindings$/}).locator('..');await detail.locator('pre').evaluate(el=>el.scrollTop=el.scrollHeight);await detail.scrollIntoViewIfNeeded();await page.screenshot({path:path.join(art,'native-copular-durable-bindings.png'),fullPage:true});
  const evidence=await control('evidence');assert.equal(evidence.parserCalls,1);assert.equal(evidence.inspectionCount,1);for(const key of ['modelCalls','configuredModelCount','automaticIntake','reviewIntakeCalls','analysisReteFacts','analysisBlackboardEntries'])assert.equal(evidence[key],0,key);assert.equal(parses().length,1);assert.equal(authored().length,1);assert.deepEqual(errors,[]);assert.deepEqual(external,[]);const result={result:'PASS',scope:'One fresh actual CoreNLP4.6 inspection from native228byte synthetic source drop, productioncatalog inert8BGP diagram and one actual connected current copular-query native graph, exact3syntaxrows/fullcopyIDs/sourcecontext/durableevidence; no inference or positiveforce claim',meta,observations,evidence,writes,browserErrors:errors,externalBrowserRequests:external};fs.writeFileSync(path.join(art,'browser-result.json'),JSON.stringify(result,null,2));console.log(JSON.stringify({result:'PASS',parserCalls:1,authoredRuns:1,copularRows:bindings.length,inspectionCid:inspection.receiptCid,runReceiptCid:run.receiptCid}));
 }catch(error){persist();fs.writeFileSync(path.join(art,'browser-continuation-failure.json'),JSON.stringify({error:String(error),stack:error.stack},null,2));if(page)await page.screenshot({path:path.join(art,'browser-continuation-failure.png'),fullPage:true}).catch(()=>{});throw error;}
 finally{if(browser)await browser.close();await control('finish','POST').catch(()=>{});}
}
main().catch(e=>{console.error(e.stack||e);process.exitCode=1;});
