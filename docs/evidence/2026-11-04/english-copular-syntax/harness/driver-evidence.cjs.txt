const fs=require('node:fs'),path=require('node:path'),assert=require('node:assert/strict'),crypto=require('node:crypto');
const {chromium}=require('playwright');
async function main(){
 const target=new URL(process.argv[2]);assert.equal(target.protocol,'http:');assert.equal(target.hostname,'127.0.0.1');
 const base=target.origin,art=__dirname,priorRaw=fs.readFileSync(path.join(art,'attempt2','browser-continuation-progress.json')),prior=JSON.parse(priorRaw),observations={...prior.observations},writes=[...prior.writes],errors=[...prior.errors],external=[...prior.external],newRequests=[],workspaceResponses=[];let browser,page;
 const hash=b=>'sha256:'+crypto.createHash('sha256').update(b).digest('hex');
 const control=async(name,method='GET')=>{const r=await fetch(base+'/_fixture/'+name,{method});assert.equal(r.status,200);return r.json();};
 const persist=()=>fs.writeFileSync(path.join(art,'browser-evidence-progress.json'),JSON.stringify({observations,writes,errors,external,newRequests,workspaceResponses},null,2));
 try{
  const meta=await control('meta'),inspection=observations.inspection,run=observations.run;assert.equal(inspection.originalCid,meta.source.cid);assert.equal(inspection.selection.project,meta.project);assert.equal(run.status,'completed');assert.equal(run.outputs.query.source.receiptCid,inspection.receiptCid);assert.equal(run.outputs.query.results.results.bindings.length,3);
  browser=await chromium.launch({headless:true,executablePath:'/usr/bin/chromium',args:['--no-sandbox']});page=await browser.newPage({viewport:{width:1920,height:1400},reducedMotion:'reduce'});
  page.on('pageerror',e=>errors.push(e.message));page.on('request',r=>{const u=new URL(r.url());if(u.origin!==base){external.push(r.url());return;}newRequests.push({path:u.pathname,method:r.method()});if(r.method()==='POST'&&!u.pathname.startsWith('/_fixture/')){let body;try{body=r.postDataJSON();}catch{body={bytes:r.postDataBuffer()?.length};}writes.push({path:u.pathname,body});}});
  page.on('response',async r=>{if(new URL(r.url()).pathname==='/api/lcnc/workspace')workspaceResponses.push({method:r.request().method(),status:r.status(),body:await r.json().catch(()=>null)});});
  const waitWorkspace=method=>page.waitForResponse(r=>new URL(r.url()).pathname==='/api/lcnc/workspace'&&r.request().method()===method,{timeout:30000});
  await page.goto(base+'/panels',{waitUntil:'domcontentloaded'});await page.locator('#presetsBtn').waitFor();await page.locator('#presetsBtn').click();await page.locator('#gallery .galRow[data-kind="preset"][data-name="preset-english-copular-query"]').click();await page.waitForFunction(()=>document.querySelectorAll('.node').length===10);
  if((await page.locator('#palette').getAttribute('class')||'').includes('open'))await page.locator('#paletteBtn').click();
  const receipt=page.locator('.node[data-id="receipt"]');await receipt.getByRole('checkbox',{name:'Use default',exact:true}).check();await receipt.locator('.params label').filter({hasText:/^default/}).locator('xpath=following-sibling::input[1]').fill(inspection.receiptCid);await page.locator('.node[data-id="mode"] .params input:not([type=checkbox])').fill('current');await page.locator('#fitBtn').click();
  const drawer=page.getByRole('region',{name:'Workspace facets'});if(!await drawer.count()){const pending=waitWorkspace('GET');await page.locator('#facetsBtn').click();await (await pending).finished();}
  const evidence=drawer.getByRole('button',{name:'Evidence',exact:true});await evidence.waitFor({timeout:30000});const pending=waitWorkspace('POST');await evidence.click();await (await pending).finished();
  const field=drawer.getByRole('textbox',{name:'Run receipt content ID',exact:true});let refreshes=0;
  for(let attempt=0;attempt<4;attempt++){
   await page.waitForFunction(()=>document.querySelector('[aria-label="Run receipt content ID"]')||[...document.querySelectorAll('[data-state]')].some(e=>e.textContent.includes('Canvas changed')),null,{timeout:30000});
   if(await field.count())break;
   const state=await drawer.locator('[data-state]').innerText();assert.match(state,/Canvas changed/);refreshes++;const pending=waitWorkspace('POST');await drawer.getByRole('button',{name:'Refresh',exact:true}).click();await (await pending).finished();
  }
  await field.waitFor({timeout:30000});await field.fill(run.receiptCid);const contentResponse=page.waitForResponse(r=>new URL(r.url()).pathname==='/api/lcnc/content'&&new URL(r.url()).searchParams.get('cid')===run.receiptCid,{timeout:30000});await drawer.getByRole('button',{name:'Inspect receipt',exact:true}).click();const actual=await (await contentResponse).json();assert.deepEqual(actual,observations.durable);await drawer.getByText(/Recorded node outputs/).waitFor({timeout:30000});assert((await drawer.innerText()).includes('Receipt: '+run.receiptCid));
  const bindings=drawer.locator('summary').filter({hasText:/^bindings$/});await bindings.click();const bindingDetail=bindings.locator('..');await bindingDetail.scrollIntoViewIfNeeded();await page.screenshot({path:path.join(art,'native-copular-durable-bindings-top.png'),fullPage:true});await bindingDetail.locator('pre').evaluate(el=>el.scrollTop=el.scrollHeight);await page.screenshot({path:path.join(art,'native-copular-durable-bindings-bottom.png'),fullPage:true});await bindings.click();
  const source=drawer.locator('summary').filter({hasText:/^source$/});await source.click();const sourceDetail=source.locator('..');await sourceDetail.scrollIntoViewIfNeeded();const sourceText=await sourceDetail.locator('pre').innerText();assert(sourceText.includes(inspection.receiptCid));assert(sourceText.includes('checked-before-and-after-query'));await sourceDetail.locator('pre').evaluate(el=>el.scrollTop=el.scrollHeight);await page.screenshot({path:path.join(art,'native-copular-durable-source.png'),fullPage:true});
  assert(!newRequests.some(r=>r.method==='POST'&&['/api/lcnc/run','/api/documents/english','/api/documents/english/query'].includes(r.path)),'read-only evidence pass must not execute a graph, inspect or query');
  observations.readonlyEvidencePass={scope:'Native gallery/current canvas loaded for comparison only. Explicit retained runreceiptCID opened through production Evidence; no new parser, upload or graph/queryexecution.',priorProgressCid:hash(priorRaw),receiptCid:run.receiptCid,refreshes,sourceText};persist();
  const finalEvidence=await control('evidence');assert.equal(finalEvidence.parserCalls,1);assert.equal(finalEvidence.inspectionCount,1);if('continuationParserCalls' in finalEvidence)assert.equal(finalEvidence.continuationParserCalls,0);for(const k of ['modelCalls','configuredModelCount','automaticIntake','reviewIntakeCalls','analysisReteFacts','analysisBlackboardEntries'])assert.equal(finalEvidence[k],0,k);assert.equal(writes.filter(w=>w.path==='/api/documents/english').length,1);assert.equal(writes.filter(w=>w.path==='/api/lcnc/run'&&w.body?.document).length,1);assert.deepEqual(errors,[]);assert.deepEqual(external,[]);
  fs.writeFileSync(path.join(art,'browser-result.json'),JSON.stringify({result:'PASS',scope:'One actual fresh CoreNLP4.6 inspection and one actual copular query across preserved attempts. Final native Evidence pass reopens exact existing receipt without execution. Syntax observations retain exact copied-node IDs and full qualifications separately in original Documents captures.',meta,observations,evidence:finalEvidence,writes,newRequests,workspaceResponses,browserErrors:errors,externalBrowserRequests:external},null,2));console.log(JSON.stringify({result:'PASS',parserCallsTotal:1,authoredRunsTotal:1,stage3NewExecutions:0,receiptCid:run.receiptCid,refreshes}));
 }catch(error){persist();fs.writeFileSync(path.join(art,'browser-evidence-failure.json'),JSON.stringify({error:String(error),stack:error.stack},null,2));if(page)await page.screenshot({path:path.join(art,'browser-evidence-failure.png'),fullPage:true}).catch(()=>{});throw error;}
 finally{if(browser)await browser.close();await control('finish','POST').catch(()=>{});}
}
main().catch(e=>{console.error(e.stack||e);process.exitCode=1;});
