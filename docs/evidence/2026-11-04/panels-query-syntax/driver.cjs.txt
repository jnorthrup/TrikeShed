const fs=require('node:fs'),path=require('node:path'),assert=require('node:assert/strict'),crypto=require('node:crypto');
const {chromium}=require('playwright');
async function main(){
 const target=new URL(process.argv[2]);assert.equal(target.protocol,'http:');assert.equal(target.hostname,'127.0.0.1');
 const base=target.origin,art=__dirname,errors=[],writes=[],external=[];let browser,page;
 const control=async name=>{const r=await fetch(base+'/_fixture/'+name,{method:name==='finish'?'POST':'GET'});assert.equal(r.status,200,name);return r.json();};
 const authored=()=>writes.filter(w=>w.path==='/api/lcnc/run'&&w.body.document);
 try{
  const meta=await control('meta'),programs=meta.panels.programs,observations={};
  browser=await chromium.launch({headless:true,executablePath:'/usr/bin/chromium',args:['--no-sandbox']});page=await browser.newPage({viewport:{width:1920,height:1400},reducedMotion:'reduce'});
  page.on('pageerror',e=>errors.push(e.message));page.on('request',r=>{const u=new URL(r.url());if(u.origin!==base)external.push(r.url());else if(r.method()==='POST')writes.push({path:u.pathname,body:r.postDataJSON()});});
  const load=async name=>{const before=authored().length;await page.goto(base+'/panels?load='+name,{waitUntil:'domcontentloaded'});await page.locator('.node').first().waitFor({timeout:45000});assert.equal(await page.locator('.node').count(),programs[name].nodes.length);const palette=page.locator('#palette');if((await palette.getAttribute('class')||'').includes('open'))await page.locator('#paletteBtn').click();await page.locator('#fitBtn').click();assert.equal(authored().length,before);};
  for(const name of ['legal-inner-thirty-days','sql-inspection-only']){
   await load(name);await page.waitForFunction(()=>document.getElementById('status').textContent.includes('Inspection only · Run and Live are disabled'),null,{timeout:15000});
   const data=await page.evaluate(()=>({nodes:[...document.querySelectorAll('.node')].map(n=>({type:n.dataset.type,syntax:n.dataset.querySyntax,label:n.querySelector('.query-syntax-status')?.textContent,warnings:[...n.querySelectorAll('.runtime-status')].map(e=>e.textContent),decorations:n.querySelectorAll(':scope > .seed,:scope > .emit,[data-sprite]').length,fields:[...n.querySelectorAll('.params input:not([type=checkbox]),.params textarea,.params select')].map(e=>({value:e.value,title:e.title}))})),laneFrames:document.querySelectorAll('#spine .lane').length,footer:document.getElementById('status').textContent}));
   assert.equal(data.laneFrames,0);assert.equal(await page.locator('#runBtn').isDisabled(),true);assert.equal(await page.locator('#liveBtn').isDisabled(),true);assert.equal(await page.locator('#debugBtn').isDisabled(),true);
   for(const node of data.nodes){assert.equal(node.syntax,'1');assert.equal(node.label,'Query syntax · Inspection only');assert.deepEqual(node.warnings,[]);assert.equal(node.decorations,0);for(const field of node.fields)assert.equal(field.title,field.value,'full native title must equal the authored field value');}
   assert(data.nodes.some(n=>n.fields.some(f=>f.value.length>35&&f.title===f.value)),'long predicate/table values remain inspectable');observations[name]=data;await page.screenshot({path:path.join(art,name+'-accurate-labels.png'),fullPage:true});
  }
  await load('mixed-inspection-only');assert.equal(await page.locator('#runBtn').isDisabled(),true);const syntax=page.locator('.node[data-query-syntax="1"]');assert.equal(await syntax.count(),programs['legal-inner-thirty-days'].nodes.length);assert.equal(await syntax.locator('.runtime-status').count(),0);const mixedWarning=await page.locator('.node[data-id="unbound"] .runtime-status').innerText();assert.match(mixedWarning,/Unavailable:.*(runtime|owner)/i);assert.equal(await page.locator('.node[data-id="unbound"] .query-syntax-status').count(),0);observations.mixed={syntaxNodes:await syntax.count(),unboundExecutableWarning:mixedWarning,runDisabled:true};await page.locator('.node[data-id="unbound"]').screenshot({path:path.join(art,'mixed-real-executable-warning-preserved.png')});
  await load('runtime-unbound-control');assert.equal(await page.locator('#runBtn').isDisabled(),false);assert.equal(await page.locator('.query-syntax-status').count(),0);const warning=await page.locator('.node[data-id="unbound"] .runtime-status').innerText();assert.match(warning,/Unavailable:.*(runtime|owner)/i);observations.unbound={warning,runDisabled:false};await page.screenshot({path:path.join(art,'executable-owner-warning-unchanged.png'),fullPage:true});assert.equal(authored().length,0,'presentation/control inspections must remain inert');
  await load('legal-known-current-a');assert.equal(await page.locator('.query-syntax-status').count(),0);assert.equal(await page.locator('#runBtn').isDisabled(),false);const waiting=page.waitForResponse(r=>new URL(r.url()).pathname==='/api/lcnc/run'&&r.request().method()==='POST'&&r.request().postDataJSON().document,{timeout:90000});await page.locator('#runBtn').click();const response=await waiting,run=await response.json();assert.equal(response.status(),200);assert.equal(run.status,'completed');assert.equal(run.ok,true);const query=run.outputs.query;assert.equal(query.dataset,'english-legal-candidate-observations');assert.equal(query.results.results.bindings[0].quote.value,'within thirty (30) business days');assert.equal(query.source.reviewStatus,'unreviewed');assert.equal(query.source.mode,'current');assert.equal(query.source.currentness,'checked-before-and-after-query');assert.equal(query.source.producerAuthority,'not-established');assert.equal(query.source.sameSentenceJoin,'cooccurrence-only-not-applicability');assert.deepEqual(run.outputs.bindings.x,query.results);assert.deepEqual(run.outputs.source.x,query.source);
  const storedResponse=await fetch(base+'/api/lcnc/content?cid='+encodeURIComponent(run.receiptCid));assert.equal(storedResponse.status,200);const raw=Buffer.from(await storedResponse.arrayBuffer());assert.equal('sha256:'+crypto.createHash('sha256').update(raw).digest('hex'),run.receiptCid);const durable=JSON.parse(raw);assert.equal(durable.lcncRun.runId,run.runId);assert.deepEqual(durable.lcncRun.outputs.query,query);fs.writeFileSync(path.join(art,'durable-receipt.json'),raw);
  await page.waitForFunction(()=>document.getElementById('status').textContent.includes('Program completed'),null,{timeout:30000});const drawer=page.getByRole('region',{name:'Workspace facets'});await drawer.waitFor({timeout:30000});await drawer.getByRole('button',{name:'Refresh',exact:true}).click();await drawer.getByRole('button',{name:'Inspect receipt',exact:true}).click();await drawer.getByText(/Recorded node outputs/).waitFor({timeout:30000});await drawer.locator('summary').filter({hasText:/^query$/}).click();await drawer.getByText('Receipt: '+run.receiptCid,{exact:false}).scrollIntoViewIfNeeded();await page.screenshot({path:path.join(art,'outer-query-still-completes-durable-evidence.png'),fullPage:true});
  const evidence=await control('evidence');assert.equal(evidence.setupQueries,0);assert.equal(evidence.safety.parserCalls,0);assert.equal(evidence.safety.reteFacts,0);assert.equal(evidence.safety.blackboardValues,0);assert.equal(authored().length,1);assert.deepEqual(errors,[]);assert.deepEqual(external,[]);
  const result={result:'PASS',scope:'Actual new immutable Panels bundle; SQL/SPARQL grammar cards in inspection-only constructions are labelled as syntax with no false runtime/agent decoration, native full-value titles and disabled execution controls. Real executable missing-owner warnings remain. One actual outer legal query still completes with durable/source evidence. Retained v58 observations only; zero new parser calls.',meta,observations,run,durable,evidence,writes,browserErrors:errors,externalBrowserRequests:external};fs.writeFileSync(path.join(art,'result.json'),JSON.stringify(result,null,2));console.log(JSON.stringify({result:'PASS',sqlNodes:observations['sql-inspection-only'].nodes.length,sparqlNodes:observations['legal-inner-thirty-days'].nodes.length,outerReceiptCid:run.receiptCid,parserCalls:0}));
 }catch(error){fs.writeFileSync(path.join(art,'failure-observations.json'),JSON.stringify({error:String(error),writes,errors,external},null,2));if(page)await page.screenshot({path:path.join(art,'failure.png'),fullPage:true}).catch(()=>{});throw error;}
 finally{if(browser)await browser.close();await control('finish').catch(()=>{});}
}
main().catch(error=>{console.error(error.stack||error);process.exitCode=1;});
