function conserve(value){
 const m=value.topics;assert.equal(m.matrixCid,value.matrixCid);assert.equal(m.corpusCid,value.corpusCid);assert.deepEqual(m.vocabulary,value.matrix.vocabulary);
 assert.equal(sum(m.topicTotals),value.matrix.tokenCount);assert.equal(m.tokenCount,value.matrix.tokenCount);
 for(let i=0;i<value.matrix.documents.length;i++)assert.equal(sum(m.documentTopicCounts[i]),value.matrix.documents[i].includedTokens);
 for(let i=0;i<value.terms.length;i++)assert.equal(sum(m.topicWordCounts.map(t=>t[i])),value.terms[i].frequency);
 for(let i=0;i<m.topicWordCounts.length;i++)assert.equal(sum(m.topicWordCounts[i]),m.topicTotals[i]);
 assert.equal(m.authority,'descriptive-statistics-only');assert.equal(m.countsAreEvidenceConfidence,false);
 for(const d of m.documentTopics)assert(Math.abs(sum(d.probabilities)-1)<1e-10);
}
