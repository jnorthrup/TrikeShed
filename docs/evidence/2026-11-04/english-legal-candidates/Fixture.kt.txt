data class FixtureSource(val id: String, val text: String) {
    val bytes = text.encodeToByteArray()
    val cid = ContentId.of(bytes)
    fun toMap() = _m["id" j id, "text" j text, "cid" j cid.value]
}

/** Synthetic localhost transport forwards production owners; no production overlays or fake NLP. */
fun main() = runBlocking {
    val artifacts = File(requireNotNull(java.lang.System.getenv("PROOF_ROOT")))
    val root = File(artifacts, "runtime").apply { require(!exists()); mkdirs() }
    val project = "english-legal-candidates-proof"
    val copyText = "The requirements for sellers and for buyers are clear."
    val sources = s_[FixtureSource("synthetic matching duties.txt",
        "Tenant shall not pay rent within thirty (30) business days unless Owner consents.\r\n" +
        "Tenant shall pay Owner.\r\n" + "Owner shall pay Tenant.\r\n"),
        FixtureSource("synthetic conflicting duties.txt",
        "Tenant shall pay rent within thirty (60) business days.\r\n" +
        "Rent shall be paid by Tenant within 30 calendar days.\r\n" +
        "Tenant shall inspect and preserve the records within two months.\r\n"),
        FixtureSource("synthetic parser copy.txt", copyText)]
    val moduleRoot = File(requireNotNull(java.lang.System.getenv("TRIKESHED_SUBVM_HOME")), "corenlp")
    val moduleManifest = File(moduleRoot, "MANIFEST.tsv").readBytes()
    check(moduleManifest.decodeToString().contains("stanford-corenlp:4.6.0"))
    val manifestCid = ContentId.of(moduleManifest)
    val files = JvmFileOperations(); val cas = PackCasStore(files, File(root, "cas").absolutePath)
    val store = CouchStoreFactory.casBacked(cas); val gateway = CouchAttachmentGateway(store, cas)
    val registry = ProjectDbRegistry("primary")
    val scopes = ProjectScopes(files, gateway, CouchIndexBridge(gateway, MemoryIndexLayer(MemoryStore(cas, store))), cas,
        null, projectDbs = registry, ledgerFile = File(root, "projects.tsv"), filesRoot = File(root, "files"))
    val miner = ProjectMiner(registry, scopes, cas, null, File(root, "files"))
    val corpus = JvmProjectCorpus(registry, scopes, miner)
    val owner = SupervisorJob(coroutineContext[Job]); val scope = CoroutineScope(Dispatchers.IO + owner + files)
    val ctx = ModuleContext(Couch("english-legal-proof", store, cas), ReteNetwork(), ReteProductionRegistry(), null, null,
        ConfixBlackboard.empty(), cas, gateway, ModuleRouteRegistry(), scope, { 123L }, root)
    val parserCalls = AtomicInteger()
    val wire = DocumentEnglishWire(ctx, corpus) {
        object : NlpReader, AutoCloseable {
            val actual = CoreNlpRuntime(CoreNlpRuntime.Profile.SYNTAX)
            override suspend fun read(text: String): NlpDocument { parserCalls.incrementAndGet(); return actual.read(text) }
            override fun close() = actual.close()
        }
    }; wire.attach()
    val automatic = AtomicInteger(); scopes.onDocument = { _, _, _ -> automatic.incrementAndGet() }
    val reviewIntake = AtomicInteger(); scopes.onTextForReview = { reviewIntake.incrementAndGet() }; scopes.onExplicitDocument = { _, _ -> reviewIntake.incrementAndGet() }
    val modelCalls = AtomicInteger()
    val brain = object : BrainClient(endpointSpecs = emptyList()) {
        override suspend fun chat(messages: List<Pair<String, String>>, maxTokens: Int, temperature: Double, contextId: String?): String {
            modelCalls.incrementAndGet(); error("No model is configured in this proof")
        }
        override suspend fun chat(options: BrainChatOptions, messages: List<Pair<String, String>>, contextId: String?): Pair<String, String> {
            modelCalls.incrementAndGet(); error("No model is configured in this proof")
        }
        override suspend fun chatExact(options: BrainChatOptions, messages: List<Pair<String, String>>, contextId: String?): Pair<String, String> {
            modelCalls.incrementAndGet(); error("No model is configured in this proof")
        }
        override suspend fun chatSeat(messages: List<Pair<String, String>>, maxTokens: Int, temperature: Double, contextId: String?, preferredModel: String?, timeoutMs: Long): Pair<String, String> {
            modelCalls.incrementAndGet(); error("No model is configured in this proof")
        }
    }
    check(brain.modelMux().listModels().size == 0 && !brain.hasEndpoints())
    val patch = PatchWire(brain, scopes, attachments = gateway, corpus = corpus, miner = miner)
    scopes.beginUpload("existing-documents", manualIntake = true)
    scopes.uploadPut("existing-documents", "other.txt", "Another selected document.".encodeToByteArray(), null, manualIntake = true)
    val inputs = File(artifacts, "drop-input").apply { mkdirs() }
    val sourceInputs = mutableListOf<String>()
    for (source in sources.view) sourceInputs.add(File(inputs, source.id).apply { writeBytes(source.bytes) }.absolutePath)
    val uploads = ProjectDbWire(registry, scopes)
    val finish = CountDownLatch(1); val held = AtomicReference(CountDownLatch(1)); val hold = AtomicReference<CountDownLatch?>(null)
    val inspections = java.util.Collections.synchronizedList(mutableListOf<Map<*, *>>())
    val statistics = java.util.Collections.synchronizedList(mutableListOf<Map<*, *>>())
    val refusals = java.util.Collections.synchronizedList(mutableListOf<Map<*, *>>())
    val lcncCalls = AtomicInteger()
    val bundle = File(requireNotNull(java.lang.System.getenv("DOCUMENTS_BUNDLE")))
    val web = File(requireNotNull(java.lang.System.getenv("FORGE_WEB_ROOT")))
    val executor = Executors.newFixedThreadPool(8)
    fun deniedNetwork(): Map<String, Any?> {
        fun denied(operation: () -> Unit): String = try { operation(); error("Outbound operation unexpectedly allowed") }
            catch (failure: java.net.SocketException) {
                check(failure.message.orEmpty().contains("Permission denied", true)) { "Unexpected outbound failure: ${failure.message}" }
                failure.message.orEmpty()
            }
        val tcp = denied { Socket().use { it.connect(InetSocketAddress("127.0.0.1", 9), 1000) } }
        val udp = denied { DatagramSocket().use { it.send(DatagramPacket(_a[1.toByte()], 1, InetAddress.getByName("127.0.0.1"), 9)) } }
        return _m["tcpConnect" j tcp, "udpSend" j udp, "result" j "DENIED"]
    }
    val denyBefore = deniedNetwork()
    val server = HttpServer.create(InetSocketAddress("127.0.0.1", 0), 8); server.executor = executor
    server.createContext("/") { exchange ->
        try {
            val method = exchange.requestMethod; val path = exchange.requestURI.rawPath; val target = exchange.requestURI.toASCIIString()
            val raw = exchange.requestBody.readBytes(); val body = raw.decodeToString()
            var status = 200; var mime = "application/json; charset=utf-8"
            val bytes = when {
                path == "/_fixture/meta" -> jsonOf(_m["project" j project, "sources" j (sources α { it.toMap() }).toList(),
                    "copyText" j copyText, "startProject" j "existing-documents", "startId" j "other.txt",
                    "nativeFilePaths" j sourceInputs, "moduleManifestCid" j manifestCid.value, "sdkVersion" j "4.6.0", "parserCounting" j "External reader delegate counts calls to the actual owned CoreNlpRuntime; no parser replacement"]).encodeToByteArray()
                path == "/_fixture/counters" -> jsonOf(_m["parserCalls" j parserCalls.get(), "inspectionCount" j inspections.size, "candidateResponses" j statistics.size]).encodeToByteArray()
                path == "/_fixture/arm" -> { held.set(CountDownLatch(1)); hold.set(CountDownLatch(1)); "{}".encodeToByteArray() }
                path == "/_fixture/held" -> jsonOf(_m["held" j (held.get().count == 0L)]).encodeToByteArray()
                path == "/_fixture/release" -> { hold.getAndSet(null)?.countDown(); "{}".encodeToByteArray() }
                path == "/_fixture/mutate" -> {
                    val before = runBlocking { requireNotNull(corpus.document(project, sources[0].id)) }
                    scopes.uploadPut(project, sources[0].id, (sources[0].text + "Amended source; retained snapshots remain historical.\r\n").encodeToByteArray(), null, manualIntake = true)
                    val after = runBlocking { requireNotNull(corpus.document(project, sources[0].id)) }
                    check(after.cid != before.cid && after.rev != before.rev && after.seq != before.seq)
                    jsonOf(after.toMap()).encodeToByteArray()
                }
                path == "/_fixture/lcnc" -> {
                    val fields = reifyStrict(body) as Map<*, *>
                    require(fields.keys == _s["receiptCid", "mode"])
                    val program = LcncProgram("retained-legal-language-proof", s_[
                        LcncNode("receipt", LcncContracts.SCOPE_IN, _m["name" j "receipt", "kind" j "id", "default" j (fields["receiptCid"] as String)]),
                        LcncNode("mode", "text.value", _m["value" j (fields["mode"] as String)]),
                        LcncNode("patterns", EnglishLegalPatternsWire.TYPE),
                        LcncNode("derived", LcncContracts.SCOPE_OUT, _m["name" j "derived", "kind" j "id"]),
                    ], s_[LcncWire("receipt", "value", "patterns", "receiptCid"), LcncWire("mode", "value", "patterns", "mode"),
                        LcncWire("patterns", "receiptCid", "derived", "value")])
                    val errors = LcncTypeCheck.check(program, LcncContracts.all().associateBy { it.type }, strict = true)
                    check(errors.isEmpty()) { errors.toString() }
                    val executed = runBlocking { LcncRunner(PureNodes.registry { 123L } + ctx.lcncRunners).runProcedure(program) }
                    val output = executed.nodeOutputs.getValue("patterns")
                    lcncCalls.incrementAndGet(); statistics.add(output)
                    jsonOf(_m["program" j reifyStrict(LcncProgramConfix.toJson(program)), "typeErrors" j errors.map { it.toString() }, "output" j output]).encodeToByteArray()
                }
                path == "/_fixture/evidence" -> {
                    for (source in sources.view) check(cas.getBounded(source.cid, source.bytes.size)?.contentEquals(source.bytes) == true)
                    val observed = synchronized(inspections) { inspections.toList() }
                    check(observed.size == 3 && parserCalls.get() == 3)
                    val payloads = synchronized(statistics) { statistics.toList() }
                    check(payloads.size >= 6)
                    val retained = mutableListOf<Map<String, Any?>>()
                    val exported = File(artifacts, "cas-readbacks").apply { mkdirs() }
                    for (payload in observed + payloads) {
                        val cid = ContentId(payload["receiptCid"] as String)
                        val block = requireNotNull(cas.getBounded(cid, 8 * 1024 * 1024)); check(ContentId.of(block) == cid)
                        val decoded = CanonicalCbor.decodeMapBounded(block, 8 * 1024 * 1024, 32, 500_000)
                        check(decoded["schema"] in _s["trikeshed.english-inspection.v1", "trikeshed.english-legal-patterns.v1"] && decoded["reviewStatus"] == "unreviewed")
                        if (decoded["schema"] == "trikeshed.english-legal-patterns.v1") check(decoded["inspectionCid"] == payload["inspectionCid"])
                        File(exported, cid.hex + ".cbor").writeBytes(block)
                        retained.add(_m["receiptCid" j cid.value, "schema" j decoded["schema"], "inspectionCid" j decoded["inspectionCid"], "bytes" j block.size])
                    }
                    for (reading in observed) {
                        val runtime = (reading["parser"] as Map<*, *>)["runtime"] as Map<*, *>
                        check(runtime["packageVersion"] == "4.6.0" && runtime["manifestCid"] == manifestCid.value)
                    }
                    check(automatic.get() == 0 && modelCalls.get() == 0 && reviewIntake.get() == 0 && lcncCalls.get() == 3)
                    check(ctx.beliefBag == null && ctx.documentReview == null && ctx.muxContext[HtxKey] == null)
                    val reteFacts = runBlocking { ctx.rete.snapshot().size }; check(reteFacts == 0)
                    val failures = synchronized(refusals) { refusals.toList() }; check(failures.isNotEmpty())
                    check(!File(root, "documents/review.wal").exists())
                    jsonOf(_m["originalSourcesRetained" j true, "inspectionCount" j observed.size, "candidateResponses" j payloads.size, "parserCalls" j parserCalls.get(),
                        "retainedReadbacks" j retained, "refusals" j failures, "lcncCalls" j lcncCalls.get(),
                        "modelCalls" j modelCalls.get(), "configuredModelCount" j brain.modelMux().listModels().size,
                        "automaticIntake" j automatic.get(), "reviewIntakeCalls" j reviewIntake.get(), "reteFacts" j reteFacts,
                        "beliefBagPresent" j false, "reviewOwnerPresent" j false, "htxPresent" j false,
                        "uploadMining" j miner.progress(project)?.let { progress -> _m["total" j progress.total, "extracted" j progress.extracted, "minted" j progress.minted, "failed" j progress.failed, "done" j progress.done] },
                        "networkDenyBefore" j denyBefore, "networkDenyAfter" j deniedNetwork()]).encodeToByteArray()
                }
                path == "/_fixture/finish" -> { finish.countDown(); "{}".encodeToByteArray() }
                path == "/documents" -> { mime = "text/html; charset=utf-8"; File(web, "documents.html").readBytes() }
                path == "/kotlin/documents/documents.js" -> { mime = "application/javascript"; bundle.readBytes() }
                path.endsWith(".css") -> { mime = "text/css"; File(web, path.substringAfterLast('/')).readBytes() }
                path.startsWith("/api/documents/english") -> {
                    if (path == EnglishLegalPatternsWire.PATH && method == "POST") hold.get()?.let { gate -> held.get().countDown(); check(gate.await(90, TimeUnit.SECONDS)) }
                    val response = runBlocking { requireNotNull(ctx.routes.match(path)).route(method, target, body, null) }!!
                    status = response.status; mime = response.contentType
                    if (method == "POST" && status == 200) {
                        val reading = reifyStrict(response.body) as Map<*, *>
                        if (path == EnglishLegalPatternsWire.PATH) statistics.add(reading) else if (path == DocumentEnglishWire.PATH) inspections.add(reading)
                    } else if (method == "POST" && status >= 400) refusals.add(_m["path" j path, "status" j status, "body" j reifyStrict(response.body)])
                    response.body.encodeToByteArray()
                }
                path.startsWith("/_project/") -> {
                    val response = runBlocking { requireNotNull(uploads.route(method, target, "$method $target HTTP/1.1\r\n\r\n".encodeToByteArray() + raw, null)) }
                    status = response.status; mime = response.contentType; response.body.encodeToByteArray()
                }
                path.startsWith("/api/projects") -> {
                    val response = runBlocking { patch.route(method, target, "$method $target HTTP/1.1\r\n\r\n$body", null) }
                    if (response == null) { status = 404; "{}".encodeToByteArray() }
                    else { status = response.status; mime = response.contentType; response.body.encodeToByteArray() }
                }
                else -> { status = 404; "{}".encodeToByteArray() }
            }
            exchange.responseHeaders.set("Content-Type", mime); exchange.sendResponseHeaders(status, bytes.size.toLong()); exchange.responseBody.use { it.write(bytes) }
        } catch (error: Throwable) {
            runCatching { val bytes = jsonOf(_m["error" j (error.message ?: "Fixture failed")]).encodeToByteArray(); exchange.sendResponseHeaders(500, bytes.size.toLong()); exchange.responseBody.use { it.write(bytes) } }
        } finally { exchange.close() }
    }
    try { server.start(); println("ENGLISH_LEGAL_PROOF_URL=http://127.0.0.1:${server.address.port}"); check(withContext(Dispatchers.IO) { finish.await(10, TimeUnit.MINUTES) }) }
    finally { hold.getAndSet(null)?.countDown(); server.stop(0); executor.shutdownNow(); wire.drain(); scopes.drain(); owner.complete(); owner.join(); cas.close()
        File(artifacts, "cleanup.json").writeText(jsonOf(_m["wireDrained" j true, "scopesDrained" j true, "supervisorCompleted" j owner.isCompleted, "parserCalls" j parserCalls.get()])) }
}
