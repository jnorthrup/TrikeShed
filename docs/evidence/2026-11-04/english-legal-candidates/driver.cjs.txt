   for(const action of sentence.modalActions){actions++;node(action.modal);node(action.action);binding(action.dependency);assert.equal(action.dependency.governorNode,action.action.id);assert.equal(action.dependency.dependentNode,action.modal.id);assert.equal(action.dependency.relation,'aux');for(const role of ['subjects','objects'])for(const item of action[role]){node(item.node);binding(item.dependency);assert.equal(item.dependency.governorNode,action.action.id);assert.equal(item.dependency.dependentNode,item.node.id);}action.relatedDependencies.forEach(binding);assert(action.unresolved.includes('legal-force-not-established'));}

  }
  assert.equal(value.counts.modalActions,actions);assert.equal(value.counts.durations,durations);for(const field of ['rules','facts','admitted','approved'])assert(!Object.hasOwn(value,field));
 };

  await page.goto(base+'/documents#'+encodeURIComponent(meta.startProject)+'/'+encodeURIComponent(meta.startId),{waitUntil:'domcontentloaded'});await page.getByRole('heading',{name:meta.startId,exact:true}).waitFor({timeout:30000});const selection=page.url();if(!await page.locator('#ds-upload').evaluate(el=>el.open))await page.locator('#ds-upload > summary').click();await page.locator('#ds-upload-project').fill(meta.project);
