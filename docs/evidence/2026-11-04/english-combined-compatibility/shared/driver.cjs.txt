const fs=require('node:fs'),path=require('node:path'),assert=require('node:assert/strict'),crypto=require('node:crypto');
const {chromium}=require('playwright');
async function main(){
 const target=new URL(process.argv[2]);assert.equal(target.protocol,'http:');assert.equal(target.hostname,'127.0.0.1');
 const base=target.origin,art=__dirname,errors=[],writes=[],external=[],results={};let browser,page;
 const control=async(name,method='GET')=>{const r=await fetch(base+'/_fixture/'+name,{method});assert.equal(r.status,200,name);return r.json();};
 const hash=b=>'sha256:'+crypto.createHash('sha256').update(b).digest('hex');
 const authored=()=>writes.filter(w=>w.path==='/api/lcnc/run'&&w.body?.document);
 const save=()=>fs.writeFileSync(path.join(art,'progress.json'),JSON.stringify({results,writes,errors,external},null,2));
 try{
  const meta=await control('meta'),docs=meta.documents,ids=docs.corpusReceiptIds;
  browser=await chromium.launch({headless:true,executablePath:'/usr/bin/chromium',args:['--no-sandbox']});page=await browser.newPage({viewport:{width:1920,height:1400},reducedMotion:'reduce'});
  // Existing production shelf restoration, containing IDs only; this is an explicit proof precondition.
  await page.addInitScript(({project,ids})=>localStorage.setItem('forge.documents.englishCorpus.v1:'+encodeURIComponent(project),JSON.stringify(ids)),{project:docs.corpusProject,ids});
  page.on('pageerror',e=>errors.push(e.message));page.on('request',r=>{const u=new URL(r.url());if(u.origin!==base)external.push(r.url());else if(r.method()==='POST'){let body;try{body=r.postDataJSON();}catch{body={binaryBytes:r.postDataBuffer()?.length};}writes.push({path:u.pathname,query:u.search,body});}});
  const row=meta.sourceRows[0];await page.goto(base+'/documents#'+encodeURIComponent(row.project)+'/'+encodeURIComponent(row.id));
  const corpus=page.locator('[data-english-corpus]');await corpus.waitFor({timeout:45000});await corpus.locator('summary').click();assert.equal(await corpus.locator('[data-corpus-selected-id]').count(),ids.length);
  const build=async()=>{const promise=page.waitForResponse(r=>new URL(r.url()).pathname==='/api/documents/english/corpus'&&r.request().method()==='POST',{timeout:90000});await corpus.locator('[data-corpus-build]').click();const response=await promise;assert.equal(response.status(),200);const out=await response.json();assert.equal(out.source.mode,'snapshot');assert.equal(out.source.historical,true);assert.equal(out.source.parserExecution,'not-reverified');assert.deepEqual(out.receiptCids,[...ids].sort());await corpus.locator('[data-corpus-result]').waitFor();return out;};
  const plain=await build();assert.equal(plain.topics,null);assert.equal(plain.rawTokenCount,70);assert.equal(plain.terms.find(t=>t.lemma==='requirement').frequency,1);await corpus.locator('[data-corpus-wordcloud]').scrollIntoViewIfNeeded();await page.screenshot({path:path.join(art,'corpus-wordcloud.png'),fullPage:true});
  await corpus.locator('[data-corpus-term="not"]').first().click();const concordance=corpus.locator('[data-corpus-concordance="not"]');await concordance.waitFor();const quotes=await concordance.locator('blockquote').allTextContents();assert(quotes.includes('Tenant shall not pay rent within thirty (30) business days unless Owner consents.'));await concordance.scrollIntoViewIfNeeded();await page.screenshot({path:path.join(art,'corpus-exact-concordance.png'),fullPage:true});
  await corpus.locator('[data-corpus-topics-enable]').check();const lda=await build();assert.equal(lda.topics.schema,'trikeshed.english-lda.v1');assert.equal(lda.topics.authority,'descriptive-statistics-only');assert.equal(lda.topics.countsAreEvidenceConfidence,false);assert.equal(lda.matrixCid,plain.matrixCid);assert.equal(lda.topics.topicTotals.reduce((a,b)=>a+b,0),plain.matrix.tokenCount);await corpus.locator('[data-corpus-topics]').scrollIntoViewIfNeeded();await page.screenshot({path:path.join(art,'corpus-local-lda.png'),fullPage:true});results.corpus={plain,lda,quotes,shelfRestoration:'explicit fixture precondition; exact historical IDs only'};save();
  const before=page.url();if(!await page.locator('#ds-upload').evaluate(el=>el.open))await page.locator('#ds-upload > summary').click();await page.locator('#ds-upload-project').fill('v63-native-upload');
  const nativePath=path.join(art,'native-upload.txt');fs.writeFileSync(nativePath,'Synthetic native batch upload; no parser or model.\r\n');
  await page.locator('#ds-upload-files').setInputFiles(nativePath);await page.locator('#ds-upload-status').getByRole('link',{name:'Open uploaded project'}).waitFor({timeout:45000});
  assert.equal(page.url(),before,'upload keeps selected document');const uploadText=await page.locator('#ds-upload-status').innerText();assert.match(uploadText,/saved.*parsing could not start|Files were saved/i);await page.locator('#ds-upload-status').getByRole('link').click();await page.locator('[data-doc]').filter({hasText:'native-upload.txt'}).first().click();await page.getByText('Synthetic native batch upload; no parser or model.',{exact:false}).first().waitFor();assert(writes.some(w=>w.path.endsWith('/putDocumentBatch')));results.upload={status:uploadText,url:page.url(),scope:'actual batch storage and navigation; real missing-miner 503, no fresh parse'};await page.screenshot({path:path.join(art,'native-batch-upload-navigation.png'),fullPage:true});save();
  await page.goto(base+'/panels?load=node-runtime-unavailable-control',{waitUntil:'domcontentloaded'});await page.locator('.node[data-id="node-provider"] .runtime-status').waitFor({timeout:45000});const nodeWarning=await page.locator('.node[data-id="node-provider"] .runtime-status').innerText();assert.match(nodeWarning,/Unavailable:.*(runtime|owner)/i);assert.equal(authored().length,0);results.browserNodeCapability={scope:'This reduced actual Panels host binds no VM runtime; no Node provider execution or sandbox claim',warning:nodeWarning};await page.screenshot({path:path.join(art,'browser-vm-owner-unavailable.png'),fullPage:true});save();
  const execute=async(name,expectedStatus=200)=>{const count=authored().length;await page.goto(base+'/panels?load='+name,{waitUntil:'domcontentloaded'});await page.locator('.node').first().waitFor({timeout:45000});assert.equal(await page.locator('.node').count(),meta.panels.programs[name].nodes.length);assert.equal(authored().length,count,'loading is inert');await page.locator('#fitBtn').click();const wait=page.waitForResponse(r=>new URL(r.url()).pathname==='/api/lcnc/run'&&r.request().method()==='POST'&&r.request().postDataJSON()?.document,{timeout:90000});await page.locator('#runBtn').click();const response=await wait,run=await response.json();assert.equal(response.status(),expectedStatus);assert.equal(run.status,expectedStatus===200?'completed':'failed');const stored=await fetch(base+'/api/lcnc/content?cid='+encodeURIComponent(run.receiptCid));assert.equal(stored.status,200);const raw=Buffer.from(await stored.arrayBuffer());assert.equal(hash(raw),run.receiptCid);const durable=JSON.parse(raw);assert.deepEqual(durable.lcncRun.outputs,run.outputs);fs.mkdirSync(path.join(art,'durable-readbacks'),{recursive:true});fs.writeFileSync(path.join(art,'durable-readbacks',run.receiptCid.slice(7)+'.json'),raw);return {run,durable};};
  results.camel=await execute('preset-english-camel-query');const camel=results.camel.run.outputs;assert.equal(camel.dispatch.routed,meta.inspections.matching);assert.equal(camel.dispatch.receipt.reply,meta.inspections.matching);assert.equal(camel.dispatch.receipt.headers.EvidenceCid,meta.inspections.matching);assert.equal(camel.query.dataset,'english-receipt-observations');assert(camel.query.results.results.bindings.some(b=>b.quote.value==='not'));assert.deepEqual(camel.bindings.x,camel.query.results);assert.deepEqual(camel.source.x,camel.query.source);save();
  await page.waitForFunction(()=>document.getElementById('status').textContent.includes('Program completed'));await page.screenshot({path:path.join(art,'actual-camel-preset.png'),fullPage:true});
  results.legal=await execute('legal-known-current-a');const legal=results.legal.run.outputs.query;assert.equal(legal.results.results.bindings[0].quote.value,'within thirty (30) business days');assert.equal(legal.source.currentness,'checked-before-and-after-query');assert.equal(legal.source.reviewStatus,'unreviewed');assert.equal(legal.source.producerAuthority,'not-established');save();
  await page.waitForFunction(()=>document.getElementById('status').textContent.includes('Program completed'));const drawer=page.getByRole('region',{name:'Workspace facets'});await drawer.waitFor();await drawer.getByRole('button',{name:'Refresh',exact:true}).click();await drawer.getByRole('button',{name:'Inspect receipt',exact:true}).click();await drawer.getByText(/Recorded node outputs/).waitFor();await drawer.locator('summary').filter({hasText:/^query$/}).click();await drawer.getByText('Receipt: '+results.legal.run.receiptCid,{exact:false}).scrollIntoViewIfNeeded();await page.screenshot({path:path.join(art,'legal-native-evidence-navigation.png'),fullPage:true});
  results.mutation=await control('mutate','POST');const request={inspectionCid:meta.inspections.matching,mode:'current',program:meta.programs.thirtyDays};let r=await fetch(base+'/api/documents/english/legal-patterns/query',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify(request)});assert.equal(r.status,409);results.stale=await r.json();request.mode='snapshot';r=await fetch(base+'/api/documents/english/legal-patterns/query',{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify(request)});assert.equal(r.status,200);results.snapshot=await r.json();assert.equal(results.snapshot.source.historical,true);assert.deepEqual(results.snapshot.results,legal.results);
  results.corpusView=await require('./corpus-view-driver.cjs').run({page,base,art});save();
  const evidence=await control('evidence');assert.equal(evidence.safety.parserCalls,0);assert.equal(evidence.safety.reteFacts,0);assert.equal(evidence.documents.automaticIntake,0);assert.equal(evidence.documents.reviewIntake,0);assert.equal(evidence.documents.configuredModels,0);assert.equal(authored().length,2);assert.equal(writes.filter(w=>w.path==='/api/documents/english').length,0);assert.deepEqual(errors,[]);assert.deepEqual(external,[]);fs.writeFileSync(path.join(art,'result.json'),JSON.stringify({result:'PASS',meta,results,evidence,writes,errors,external},null,2));console.log(JSON.stringify({result:'PASS',parserCalls:0,authoredRuns:2,corpusBuilds:2}));
 }catch(error){save();fs.writeFileSync(path.join(art,'failure.json'),JSON.stringify({error:String(error),stack:error.stack},null,2));if(page)await page.screenshot({path:path.join(art,'failure.png'),fullPage:true}).catch(()=>{});throw error;}
 finally{if(browser)await browser.close();await control('finish','POST').catch(()=>{});}
}
main().catch(e=>{console.error(e.stack||e);process.exitCode=1;});
