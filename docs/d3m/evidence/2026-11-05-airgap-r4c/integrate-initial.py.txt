from pathlib import Path
import subprocess,json,datetime
repo=Path('/workspace/TrikeShed'); out=repo/'.git/d3m/airgap-interrupted-execution-20261105';out.mkdir(parents=True,exist_ok=True)
def git(*args,check=True):
    r=subprocess.run(['git',*args],cwd=repo,text=True,stdout=subprocess.PIPE,stderr=subprocess.STDOUT)
    if check and r.returncode: raise RuntimeError(r.stdout)
    return r
assert not git('status','--porcelain').stdout
packet=json.loads(Path('/workspace/airgap-r4c-proof/integration-inputs.json').read_text());refs=packet['refs']
(out/'integration-inputs.json').write_text(json.dumps(packet,indent=2)+'\n')
git('switch','master')
seen=set();steps=[]
for r in sorted(refs,key=lambda r:(r['time'],r['commit'],r['ref'])):
    if r['commit'] in seen:continue
    seen.add(r['commit'])
    if git('merge-base','--is-ancestor',r['commit'],'HEAD',check=False).returncode==0:continue
    before=git('rev-parse','HEAD').stdout.strip()
    result=git('-c','rerere.enabled=false','-c','merge.conflictStyle=diff3','merge','--no-ff','--no-edit','-m','D3M interrupted-execution milestone: '+r['ref'],r['commit'],check=False)
    step=dict(**r,before=before,exit=result.returncode,output=result.stdout);steps.append(step)
    (out/'integration-steps.json').write_text(json.dumps(steps,indent=2)+'\n')
    if result.returncode:
        index=git('ls-files','-u').stdout;(out/'conflict-index.txt').write_text(index)
        for line in index.splitlines():
            metadata,path=line.split('\t');mode,oid,stage=metadata.split();b=out/'conflicts'/path;b.parent.mkdir(parents=True,exist_ok=True)
            Path(str(b)+'.stage'+stage).write_bytes(subprocess.check_output(['git','cat-file','blob',oid],cwd=repo))
        raise RuntimeError('Conflicts retained for intent review. '+result.stdout)
    step['after']=git('rev-parse','HEAD').stdout.strip()
    (out/'integration-steps.json').write_text(json.dumps(steps,indent=2)+'\n')
head=git('rev-parse','HEAD').stdout.strip()
ancestors=set(git('rev-list',head).stdout.splitlines())
for r in refs:assert r['commit'] in ancestors,r
print(json.dumps(dict(head=head,refs=len(refs),merges=len(steps),steps=[{k:s[k] for k in ('ref','commit','after')} for s in steps]),indent=2))
