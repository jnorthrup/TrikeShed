#!/usr/bin/env python3
# Assemble the committed D3M evidence packet for d3m-20261107-191424 from the
# local archive (.git/d3m/<run>), audit conflict stage blobs against the retained
# unmerged index and the provisional carrier, and write SHA256.json last.
from pathlib import Path
import subprocess, json, hashlib, shutil

repo = Path('/Users/jim/work/TrikeShed')
run = repo / '.git/d3m/d3m-20261107-191424'
pkt = repo / 'docs/d3m/evidence/2026-11-07'
incoming = '5dad15b012473bf5cd1367eac5e74260873dcfe1'
carrier = 'd05857be235dcc49756866fec3121dfcd4b8d68a'


def git(*a, text=True):
    return subprocess.check_output(['git', *a], cwd=repo, text=text)


pkt.mkdir(parents=True, exist_ok=True)
(pkt / '.gitattributes').write_text('* -text -whitespace\n')
copies = {
    'fetch.py': 'fetch.py.txt', 'integrate.py': 'integrate.py.txt', 'resolve.py': 'resolve.py.txt',
    'verify.sh': 'verify.sh.txt', 'packet.py': 'packet.py.txt',
    'initial-remote-heads.json': None, 'initial-fetch.json': None, 'initial-inputs.json': None,
    'frozen-remote-heads.json': None, 'frozen-fetch.json': None, 'frozen-inputs.json': None,
    'integration-steps.json': None, 'intent-resolution.json': None,
    'checkpoint-notes.md': 'checkpoint-notes.txt', 'pre-checkpoint-status.txt': None,
    'casefold-phantom.diff': 'casefold-phantom.diff.txt',
    'verify-identity.txt': None, 'verify-gradle.log': 'verify-gradle.log.txt', 'verify-results.json': None,
    'integrate-final.txt': None,
    'publication-remote-heads.json': None, 'publication-fetch.json': None, 'publication-inputs.json': None,
    'integrate-continuation.py': 'integrate-continuation.py.txt', 'integrate-continuation.txt': None,
    'continuation-steps.json': None, 'verify-continuation.sh': 'verify-continuation.sh.txt',
    'verify-continuation-identity.txt': None, 'verify-continuation-gradle.log': 'verify-continuation-gradle.log.txt',
    'verify-continuation-results.json': None,
    'publication2-remote-heads.json': None, 'publication2-fetch.json': None, 'publication2-inputs.json': None,
    'integrate-continuation2.py': 'integrate-continuation2.py.txt', 'integrate-continuation2.txt': None,
    'continuation2-steps.json': None, 'verify-continuation2.sh': 'verify-continuation2.sh.txt',
    'verify-continuation2-identity.txt': None, 'verify-continuation2-gradle.log': 'verify-continuation2-gradle.log.txt',
    'verify-continuation2-results.json': None,
    f'conflict-index-{incoming}.txt': None,
}
for src, dst in copies.items():
    s = run / src
    if s.exists():
        shutil.copyfile(s, pkt / (dst or src))

# stage blobs: copy and audit against the archived unmerged index
index = (run / f'conflict-index-{incoming}.txt').read_text()
audit = []
for line in index.splitlines():
    meta, path = line.split('\t')
    mode, oid, stage = meta.split()
    rel = f'conflicts/{incoming}/{path}.stage{stage}'
    s = run / rel
    d = pkt / rel
    d.parent.mkdir(parents=True, exist_ok=True)
    shutil.copyfile(s, d)
    data = d.read_bytes()
    blob = subprocess.run(['git', 'hash-object', '--stdin'], cwd=repo, input=data,
                          stdout=subprocess.PIPE, check=True).stdout.decode().strip()
    audit.append(dict(file=rel, mode=mode, stage=int(stage), gitObjectId=oid,
                      sha256=hashlib.sha256(data).hexdigest(), matchesConflictIndex=blob == oid))
# marked working files: kept in the provisional carrier commit, audited here (not copied)
for path in sorted({l.split('\t')[1] for l in index.splitlines()}):
    marked = (run / f'conflicts/{incoming}/{path}.marked').read_bytes()
    committed = subprocess.check_output(['git', 'show', f'{carrier}:{path}'], cwd=repo)
    audit.append(dict(markedWorkingFile=path, carrier=carrier,
                      sha256=hashlib.sha256(marked).hexdigest(),
                      matchesCarrierBlob=marked == committed,
                      conflictBlocks=marked.count(b'\n<<<<<<< ') + (1 if marked.startswith(b'<<<<<<< ') else 0)))
(pkt / 'conflict-blob-audit.json').write_text(json.dumps(audit, indent=2) + '\n')
assert all(a.get('matchesConflictIndex', True) and a.get('matchesCarrierBlob', True) for a in audit), audit

files = sorted(p for p in pkt.rglob('*') if p.is_file() and p.name != 'SHA256.json')
manifest = {str(p.relative_to(pkt)): hashlib.sha256(p.read_bytes()).hexdigest() for p in files}
(pkt / 'SHA256.json').write_text(json.dumps(manifest, indent=2) + '\n')
print(json.dumps(dict(files=len(manifest), stageBlobs=sum(1 for a in audit if 'stage' in a),
                      allStagesMatch=all(a['matchesConflictIndex'] for a in audit if 'stage' in a),
                      markedMatchCarrier=all(a['matchesCarrierBlob'] for a in audit if 'carrier' in a)), indent=2))
