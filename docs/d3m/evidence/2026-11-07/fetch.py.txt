#!/usr/bin/env python3
# D3M step 1 (PRELOAD.md): enumerate every remote's heads, fetch all branch heads
# explicitly (--no-prune, refs/heads/*:refs/remotes/<remote>/*), then freeze
# local + remote ref names, commit IDs and committer timestamps.
# Usage: fetch.py <label>   (writes <label>-remote-heads.json, <label>-fetch.json,
#                            <label>-inputs.json into this run directory)
from pathlib import Path
import subprocess, json, datetime, sys

repo = Path('/Users/jim/work/TrikeShed')
out = Path(__file__).resolve().parent
label = sys.argv[1] if len(sys.argv) > 1 else 'initial'


def git(*args):
    return subprocess.check_output(['git', *args], cwd=repo, text=True)


heads = {}
fetch = {}
for remote in git('remote').splitlines():
    heads[remote] = git('ls-remote', '--heads', remote)
    p = subprocess.run(['git', 'fetch', '--no-prune', remote,
                        'refs/heads/*:refs/remotes/' + remote + '/*'],
                       cwd=repo, text=True, stdout=subprocess.PIPE, stderr=subprocess.STDOUT)
    fetch[remote] = {'exit': p.returncode, 'output': p.stdout}
    assert p.returncode == 0, p.stdout

refs = []
for line in git('for-each-ref',
                '--format=%(refname)%00%(objectname)%00%(committerdate:unix)%00%(objecttype)',
                'refs/heads', 'refs/remotes').splitlines():
    name, commit, stamp, kind = line.split('\0')
    if kind != 'commit':
        continue
    refs.append({'ref': name, 'commit': commit, 'time': int(stamp)})

# every advertised remote head must now be frozen at the advertised OID
adv_missing = []
frozen = {r['ref']: r['commit'] for r in refs}
for remote, text in heads.items():
    for l in text.splitlines():
        oid, ref = l.split('\t')
        local = 'refs/remotes/' + remote + '/' + ref[len('refs/heads/'):]
        if frozen.get(local) != oid:
            adv_missing.append({'remote': remote, 'ref': ref, 'advertised': oid, 'frozen': frozen.get(local)})

packet = {
    'recordedAt': datetime.datetime.now(datetime.timezone.utc).isoformat(),
    'head': git('rev-parse', 'HEAD').strip(),
    'branch': git('rev-parse', '--abbrev-ref', 'HEAD').strip(),
    'status': git('status', '--porcelain=v1'),
    'refs': refs,
    'advertisedNotFrozen': adv_missing,
}
(out / (label + '-remote-heads.json')).write_text(json.dumps(heads, indent=2) + '\n')
(out / (label + '-fetch.json')).write_text(json.dumps(fetch, indent=2) + '\n')
(out / (label + '-inputs.json')).write_text(json.dumps(packet, indent=2) + '\n')
print(json.dumps({'head': packet['head'], 'refs': len(refs),
                  'advertised': {k: len(v.splitlines()) for k, v in heads.items()},
                  'advertisedNotFrozen': len(adv_missing),
                  'fetch': {k: v['output'][-1500:] for k, v in fetch.items()}}, indent=2))
