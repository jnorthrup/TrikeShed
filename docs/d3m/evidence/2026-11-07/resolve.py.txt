def blocks(text):
    """Split diff3-marked text into segments: ('plain', lines) or ('conflict', ours, base, theirs)."""
    segs, cur, state, ours, base, theirs = [], [], None, [], [], []
    for line in text.splitlines(keepends=True):
        if state is None and line.startswith('<<<<<<< '):
            segs.append(('plain', cur)); cur = []; state = 'ours'; ours, base, theirs = [], [], []
        elif state == 'ours' and line.startswith('||||||| '):
            state = 'base'
        elif state in ('ours', 'base') and line.rstrip('\n') == '=======':
            state = 'theirs'
        elif state == 'theirs' and line.startswith('>>>>>>> '):
            segs.append(('conflict', ours, base, theirs)); state = None
        elif state == 'ours': ours.append(line)
        elif state == 'base': base.append(line)
        elif state == 'theirs': theirs.append(line)
        else: cur.append(line)
    assert state is None
    segs.append(('plain', cur))
    return segs


def show(rev, path):
    return subprocess.check_output(['git', 'show', f'{rev}:{path}'], cwd=repo, text=True)


record = []
resolved = {}
for path, decisions in DECISIONS.items():
    marked = (cdir / (path + '.marked')).read_text()
    segs = blocks(marked)
    conflicts = [s for s in segs if s[0] == 'conflict']
    assert len(conflicts) == len(decisions), (path, len(conflicts))
    eq_text = show(equivalent, path)
    assert eq_text == show(incoming, path), path
    text = []
    n = 0
    for s in segs:
        if s[0] == 'plain':
            text += s[1]
            continue
        _, ours, base, theirs = s
        d = decisions[n]; n += 1
        # the incoming text must appear verbatim in the already-integrated commit
        assert ''.join(theirs) in eq_text, (path, d['block'])
        text += ours
        record.append(dict(path=path, **d,
                           lines=dict(head=len(ours), base=len(base), incoming=len(theirs)),
                           sha256=dict(head=hashlib.sha256(''.join(ours).encode()).hexdigest(),
                                       base=hashlib.sha256(''.join(base).encode()).hexdigest(),
                                       incoming=hashlib.sha256(''.join(theirs).encode()).hexdigest()),
                           incomingVerbatimIn=equivalent))
    resolved[path] = ''.join(text)
    same = resolved[path] == show(premerge, path)
    record.append(dict(path=path, resolvedEqualsPremergeMaster=same))
    print(path, 'blocks', n, 'resolved == pre-merge master blob:', same)

(out / 'intent-resolution.json').write_text(json.dumps(record, indent=2) + '\n')
if '--apply' in sys.argv:
    for path, text in resolved.items():
        (repo / path).write_text(text)
    print('applied')
